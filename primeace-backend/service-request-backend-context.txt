app.use(errorHandler);

import mongoose from "mongoose";

import { env } from "./env.js";

const DATABASE_OPTIONS = {
  serverSelectionTimeoutMS: 10_000,
  autoIndex: env.NODE_ENV !== "production"
};

let connectionPromise: Promise<typeof mongoose> | null = null;

export async function connectDatabase(): Promise<void> {
  if (mongoose.connection.readyState === 1) {
    return;
  }

  if (connectionPromise) {
    await connectionPromise;
    return;
  }

  connectionPromise = mongoose.connect(
    env.MONGODB_URI,
    DATABASE_OPTIONS
  );

  try {
    await connectionPromise;

    console.log("MongoDB connected successfully.");
  } catch (error) {
    console.error("MongoDB connection failed:", error);
    throw error;
  } finally {
    connectionPromise = null;
  }
}

export async function disconnectDatabase(): Promise<void> {
  if (mongoose.connection.readyState === 0) {
    return;
  }

  await mongoose.disconnect();

  console.log("MongoDB disconnected successfully.");
}

mongoose.connection.on("disconnected", () => {
  console.warn("MongoDB connection lost.");
});

mongoose.connection.on("error", (error: Error) => {
  console.error("MongoDB connection error:", error);
});

  (value) =>
    typeof value === "string" && value.trim() === ""

      : value,

    return ["http:", "https:"].includes(url.protocol);
  },

);

const envSchema = z
  .object({
    NODE_ENV: z
      .enum(["development", "production", "test"])
      .default("development"),

    PORT: z.coerce
      .number()
      .int()

      .max(65535)
      .default(5000),

      .string()
      .trim()
      .min(1, "MONGODB_URI is required.")
      .regex(
        /^mongodb(?:\+srv)?:\/\//,
        "MONGODB_URI must be a valid MongoDB connection string."
      ),

    FRONTEND_URL: frontendUrlSchema,

    CLOUDINARY_CLOUD_NAME: optionalEnvString,
    CLOUDINARY_API_KEY: optionalEnvString,
    CLOUDINARY_API_SECRET: optionalEnvString,

    EMAIL_API_KEY: optionalEnvString,
    EMAIL_FROM: optionalEnvString
  })
  .strict()
  .superRefine((values, context) => {
    const cloudinaryValues = [
      values.CLOUDINARY_CLOUD_NAME,
      values.CLOUDINARY_API_KEY,
      values.CLOUDINARY_API_SECRET
    ];

    const cloudinaryConfigured = cloudinaryValues.some(Boolean);

        code: "custom",

        path: ["CLOUDINARY_CLOUD_NAME"]
      });

    if (Boolean(values.EMAIL_API_KEY) !== Boolean(values.EMAIL_FROM)) {
      context.addIssue({
        code: "custom",
        message:
          "EMAIL_API_KEY and EMAIL_FROM must be provided together.",
        path: ["EMAIL_API_KEY"]
      });
    }
  });

import type { RequestHandler } from "express";

import { Admin } from "../modules/auth/Admin.js";
import { AdminSession } from "../modules/auth/adminSession.js";

import {
  ADMIN_SESSION_COOKIE,
  hashSessionToken
} from "../utils/session.utils.js";

import { ApiError } from "../shared/errors/ApiError.js";

const authenticate: RequestHandler = async (
  request,
  response,
  next
) => {
  const token = request.cookies?.[ADMIN_SESSION_COOKIE];

  if (typeof token !== "string" || !token) {
    throw new ApiError(401, "Authentication is required.");
  }

  const now = new Date();

  const session = await AdminSession.findOne({
    tokenHash: hashSessionToken(token),
    revokedAt: null,
    expiresAt: { $gt: now }
  }).select("+csrfTokenHash");

  if (!session) {
    throw new ApiError(401, "Your session is invalid or has expired.");
  }

  const admin = await Admin.findOne({
    _id: session.adminId,
    isActive: true,
    isDeleted: false
  });

  if (!admin) {
    await AdminSession.updateOne(
      {
        _id: session._id,
        revokedAt: null
      },
      {
        $set: {
          revokedAt: now
        }
      }
    );

    throw new ApiError(401, "Your administrator account is unavailable.");
  }

  await AdminSession.updateOne(
    {
      _id: session._id,
      revokedAt: null
    },
    {
      $set: {
        lastUsedAt: now
      }
    }
  );

  response.locals.admin = admin;
  response.locals.adminSession = session;

  next();
};

export default authenticate;

import type { RequestHandler } from "express";

import {
  type AdminRole,
  type AdminDocument
} from "../modules/auth/Admin.js";

import { ApiError } from "../shared/errors/ApiError.js";

export function authorize(
  ...allowedRoles: AdminRole[]
): RequestHandler {
  return (_request, response, next) => {
    const admin = response.locals.admin as
      | AdminDocument
      | undefined;

    if (!admin) {
      throw new ApiError(401, "Authentication is required.");
    }

    if (!allowedRoles.includes(admin.role)) {
      throw new ApiError(
        403,
        "You do not have permission to perform this action."
      );
    }

    next();
  };
}

import type { ErrorRequestHandler } from "express";
import mongoose from "mongoose";
import { ZodError } from "zod";

import type { ApiFailure, ApiErrorDetail } from "@primeace/shared";

import { env } from "../config/env.js";
import { ApiError } from "../shared/errors/ApiError.js";

function hasErrorCode(error: unknown, code: number): boolean {
  return (
    typeof error === "object" &&
    error !== null &&
    "code" in error &&
    error.code === code
  );
}

function hasErrorType(error: unknown, type: string): boolean {
  return (
    typeof error === "object" &&
    error !== null &&
    "type" in error &&
    error.type === type
  );
}

const errorHandler: ErrorRequestHandler = (
  error,
  _request,
  response,
  next
) => {
  if (response.headersSent) {
    next(error);
    return;
  }

  let statusCode = 500;
  let message = "An unexpected error occurred.";
  let errors: ApiErrorDetail[] = [];

  if (error instanceof ApiError) {
    statusCode = error.statusCode;
    message = error.message;
    errors = error.errors;
  } else if (error instanceof ZodError) {
    statusCode = 400;
    message = "Request validation failed.";

    errors = error.issues.map((issue) => ({
      field: issue.path.join("."),
      message: issue.message,
      code: issue.code
    }));
  } else if (hasErrorType(error, "entity.parse.failed")) {
    statusCode = 400;
    message = "Request body contains invalid JSON.";
  } else if (hasErrorType(error, "entity.too.large")) {
    statusCode = 413;
    message = "Request body exceeds the permitted size.";
  } else if (error instanceof mongoose.Error.ValidationError) {
    statusCode = 400;
    message = "Database validation failed.";

    errors = Object.values(error.errors).map((issue) => ({
      field: issue.path,
      message: issue.message,
      code: "database_validation"
    }));
  } else if (error instanceof mongoose.Error.CastError) {
    statusCode = 400;
    message = "Invalid resource reference.";

    errors = [{
      field: error.path,
      message: "The supplied value is invalid.",
      code: "invalid_reference"
    }];
  } else if (hasErrorCode(error, 11000)) {
    statusCode = 409;
    message = "A record with the supplied information already exists.";
  }

  if (statusCode >= 500) {
    console.error("Application error:", error);

    if (env.NODE_ENV !== "development") {
      message = "An unexpected server error occurred.";
    } else if (error instanceof Error) {
      message = error.message;
    }
  }

  const responseBody: ApiFailure = errors.length > 0
    ? {
        success: false,
        message,
        errors
      }
    : {
        success: false,
        message
      };

  response.status(statusCode).json(responseBody);
};

export default errorHandler;

import type { RequestHandler } from "express";
import { z } from "zod";

export type ValidatedRequest<TSchema extends z.ZodType> =
  z.output<TSchema>;

export function validateRequest<TSchema extends z.ZodType>(
  schema: TSchema
): RequestHandler {
  return (request, response, next) => {
    const result = schema.safeParse({
      body: request.body,
      params: request.params,
      query: request.query
    });

    if (!result.success) {
      next(result.error);
      return;
    }

    response.locals.validated = result.data;

    next();
  };
}

import { timingSafeEqual } from "node:crypto";
import type { RequestHandler } from "express";

import type { AdminSessionDocument } from "../modules/auth/adminSession.js";

import {
  ADMIN_CSRF_COOKIE,
  hashCsrfToken
} from "../utils/session.utils.js";

import { ApiError } from "../shared/errors/ApiError.js";

const verifyCsrfToken: RequestHandler = (
  request,
  response,
  next
) => {
  const cookieToken = request.cookies?.[ADMIN_CSRF_COOKIE];
  const headerToken = request.get("x-csrf-token");

  const session = response.locals.adminSession as
    | AdminSessionDocument
    | undefined;

  if (
    typeof cookieToken !== "string" ||
    typeof headerToken !== "string" ||
    !session ||
    !session.csrfTokenHash
  ) {
    throw new ApiError(403, "CSRF validation failed.");
  }

  const cookieHash = hashCsrfToken(cookieToken);
  const headerHash = hashCsrfToken(headerToken);
  const storedHash = session.csrfTokenHash;

  const cookieBuffer = Buffer.from(cookieHash, "hex");
  const headerBuffer = Buffer.from(headerHash, "hex");
  const storedBuffer = Buffer.from(storedHash, "hex");

  const valid =
    cookieBuffer.length === storedBuffer.length &&
    headerBuffer.length === storedBuffer.length &&
    timingSafeEqual(cookieBuffer, headerBuffer) &&
    timingSafeEqual(headerBuffer, storedBuffer);

  if (!valid) {
    throw new ApiError(403, "CSRF validation failed.");
  }

  next();
};

export default verifyCsrfToken;

import type { RequestHandler } from "express";

import { env } from "../config/env.js";
import { ApiError } from "../shared/errors/ApiError.js";

const SAFE_METHODS = new Set(["GET", "HEAD", "OPTIONS"]);

const allowedOrigin = new URL(env.FRONTEND_URL).origin;

const verifyRequestOrigin: RequestHandler = (
  request,
  _response,
  next
) => {
  if (SAFE_METHODS.has(request.method)) {
    next();
    return;
  }

  const requestOrigin = request.get("origin");

  // Permit requests without an Origin header in development
  // to support local CLI tools and API testing.
  if (!requestOrigin && env.NODE_ENV !== "production") {
    next();
    return;
  }

  let normalizedRequestOrigin: string;

  try {
    normalizedRequestOrigin = new URL(requestOrigin ?? "").origin;
  } catch {
    throw new ApiError(403, "Request origin is not permitted.");
  }

  if (normalizedRequestOrigin !== allowedOrigin) {
    throw new ApiError(403, "Request origin is not permitted.");
  }

  next();
};

export default verifyRequestOrigin;

function toAuthenticatedAdmin(admin: {
  _id: unknown;
  fullName: string;
  email: string;
  role: string;
}): AuthenticatedAdmin {
  return {
    id: String(admin._id),
    fullName: admin.fullName,
    email: admin.email,
    role: admin.role

export async function loginAdmin(
  credentials: LoginCredentials,
  metadata: CreateSessionMetadata
): Promise<LoginResult> {
  const admin = await Admin.findOne({
    email: credentials.email,
    isActive: true,
    isDeleted: false
  }).select("+passwordHash");

  if (!admin) {
    throw new ApiError(401, "Invalid email or password.");
  }

  const passwordIsValid = await argon2.verify(
    admin.passwordHash,
    credentials.password
  );

  if (!passwordIsValid) {
    throw new ApiError(401, "Invalid email or password.");
  }

  const token = generateSessionToken();
  const csrfToken = generateCsrfToken();
  const expiresAt = new Date(Date.now() + ADMIN_SESSION_DURATION_MS);

  await AdminSession.create({
    adminId: admin._id,
    tokenHash: hashSessionToken(token),
    csrfTokenHash: hashCsrfToken(csrfToken),
    expiresAt,
    lastUsedAt: new Date(),
    ipAddress: metadata.ipAddress,
    userAgent: metadata.userAgent
  });

  return {
    token,
    csrfToken,
    expiresAt,
    admin: toAuthenticatedAdmin(admin)
  };
}

export async function logoutAdmin(token: string): Promise<void> {
  await AdminSession.updateOne(
    {
      tokenHash: hashSessionToken(token),
      revokedAt: null
    },
    {
      $set: {
        revokedAt: new Date()
      }
    }
  );
}

  _id: unknown;
  fullName: string;
  email: string;
  role: string;
}): AuthenticatedAdmin {

function createContractNumber(): string {
  const year = new Date().getFullYear();

  const reference = randomBytes(5)
    .toString("hex")
    .toUpperCase();

  return `PA-CON-${year}-${reference}`;

function hashToken(token: string): string {
  return createHash("sha256")
    .update(token)
    .digest("hex");
}

function validateObjectId(
  value: string,
  field: string
): Types.ObjectId {
  if (!Types.ObjectId.isValid(value)) {
    throw new ApiError(400, `Invalid ${field}.`);
  }

  return new Types.ObjectId(value);
}

function parseOptionalDate(
  value: string | null | undefined,
  field: string
): Date | null {
  if (value === undefined || value === null) {
    return null;
  }

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    throw new ApiError(400, `Invalid ${field}.`);
  }

  return date;
}

function validateContractDates(input: {
  startDate?: Date | null;
  expectedCompletionDate?: Date | null;
  acceptanceDeadline?: Date | null;
}): void {
  const {
    startDate,
    expectedCompletionDate,
    acceptanceDeadline
  } = input;

  if (
    startDate &&
    expectedCompletionDate &&
    expectedCompletionDate < startDate
  ) {
    throw new ApiError(
      400,
      "Expected completion date cannot be earlier than the start date."
    );
  }

  if (
    acceptanceDeadline &&
    acceptanceDeadline <= new Date()
  ) {
    throw new ApiError(
      400,
      "Contract acceptance deadline must be a future date."
    );
  }
}

/**
 * Create a contract draft from an approved quotation.
 */
export async function createContractDraft(
  quotationId: string,
  input: ContractDraftInput,
  adminId: string,
  context: AuditContext = {}
) {
  const quotationObjectId = validateObjectId(
    quotationId,
    "quotation ID"
  );

  const administratorId = validateObjectId(
    adminId,
    "administrator ID"
  );

  const title = input.title.trim();

  if (!title) {
    throw new ApiError(
      400,
      "Contract title is required."
    );
  }

  const startDate = parseOptionalDate(
    input.startDate,
    "contract start date"
  );

  const expectedCompletionDate = parseOptionalDate(
    input.expectedCompletionDate,
    "expected completion date"
  );

  const acceptanceDeadline = parseOptionalDate(
    input.acceptanceDeadline,
    "acceptance deadline"
  );

  validateContractDates({
    startDate,
    expectedCompletionDate,
    acceptanceDeadline
  });

  const session = await mongoose.startSession();

  let createdContractId: Types.ObjectId | null = null;

  try {
    await session.withTransaction(async () => {
      const quotation = await Quotation.findOne({
        _id: quotationObjectId,
        status: QuotationStatus.APPROVED
      }).session(session);

      if (!quotation) {
        throw new ApiError(
          404,
          "Approved quotation not found."
        );
      }

      const existingContract = await Contract.exists({
        quotationId: quotation._id
      }).session(session);

      if (existingContract) {
        throw new ApiError(
          409,
          "A contract has already been created from this quotation."
        );
      }

      if (!quotation.clientSnapshot) {
        throw new ApiError(
          400,
          "The approved quotation does not contain a valid client snapshot."
        );
      }

      if (
        quotation.items.length === 0 ||
        !Number.isSafeInteger(quotation.totalAmountKobo) ||
        quotation.totalAmountKobo < 0
      ) {
        throw new ApiError(
          400,
          "The approved quotation contains invalid contract information."
        );
      }

      const contract = new Contract({
        contractNumber: createContractNumber(),

        clientId: quotation.clientId,

        clientSnapshot: {
          fullName: quotation.clientSnapshot.fullName,
          email: quotation.clientSnapshot.email,
          phone: quotation.clientSnapshot.phone
        },

        serviceRequestId: quotation.serviceRequestId,

        quotationId: quotation._id,
        quotationNumber: quotation.quotationNumber,

        projectId: null,

        version: 1,
        previousVersionId: null,

        title,

        scope: quotation.scope,

        items: quotation.items.map((item) => ({
          serviceName: item.serviceName,
          description: item.description,
          deliverables: item.deliverables,
          quantity: item.quantity,
          unitAmountKobo: item.unitAmountKobo,
          totalAmountKobo: item.totalAmountKobo
        })),

        deliverables: [
          ...new Set(
            quotation.items.flatMap(
              (item) => item.deliverables
            )
          )
        ],

        contractAmountKobo: quotation.totalAmountKobo,

        terms: quotation.terms,

        startDate,
        expectedCompletionDate,
        acceptanceDeadline,

        status: ContractStatus.DRAFT,

        issuedAt: null
      });

      await contract.save({ session });

      await writeAuditLog(

/**
 * Update an existing contract draft.
 *
 * Only editable contract details can be changed.
 * Financial values, quotation references, version information
 * and contract status remain immutable through this operation.
 */
export async function updateContractDraft(
  contractId: string,
  input: UpdateContractInput,
  adminId: string,
  context: AuditContext = {}
) {
  const id = validateObjectId(
    contractId,
    "contract ID"
  );

  const administratorId = validateObjectId(
    adminId,
    "administrator ID"
  );

  const session = await mongoose.startSession();

  let updatedContractId: Types.ObjectId | null = null;

  try {
    await session.withTransaction(async () => {
      const contract = await Contract.findById(id)
        .session(session);

      if (!contract) {
        throw new ApiError(404, "Contract not found.");
      }

      if (contract.status !== ContractStatus.DRAFT) {
        throw new ApiError(
          409,
          "Only draft contracts can be updated."
        );
      }

      const startDate: Date | null =
        input.startDate !== undefined
          ? parseOptionalDate(
              input.startDate,
              "contract start date"
            )
          : contract.startDate ?? null;

      const expectedCompletionDate: Date | null =
        input.expectedCompletionDate !== undefined
          ? parseOptionalDate(
              input.expectedCompletionDate,
              "expected completion date"
            )
          : contract.expectedCompletionDate ?? null;

      const acceptanceDeadline: Date | null =
        input.acceptanceDeadline !== undefined
          ? parseOptionalDate(
              input.acceptanceDeadline,
              "acceptance deadline"
            )
          : contract.acceptanceDeadline ?? null;

      validateContractDates({
        startDate,
        expectedCompletionDate,
        acceptanceDeadline:
          input.acceptanceDeadline !== undefined
            ? acceptanceDeadline
            : null
      });

      const changes: Record<string, unknown> = {};

      if (input.title !== undefined) {
        changes.title = {
          before: contract.title,
          after: input.title
        };

        contract.title = input.title;
      }

      if (input.scope !== undefined) {
        changes.scope = {
          before: contract.scope,
          after: input.scope
        };

        contract.scope = input.scope;
      }

      if (input.deliverables !== undefined) {
        changes.deliverables = {
          before: [...contract.deliverables],
          after: [...input.deliverables]
        };

        contract.deliverables = [...input.deliverables];
      }

      if (input.terms !== undefined) {
        changes.terms = {
          before: [...contract.terms],
          after: [...input.terms]
        };

        contract.terms = [...input.terms];
      }

      if (input.startDate !== undefined) {
        changes.startDate = {
          before: contract.startDate,
          after: startDate
        };

        contract.startDate = startDate;
      }

      if (input.expectedCompletionDate !== undefined) {
        changes.expectedCompletionDate = {
          before: contract.expectedCompletionDate,
          after: expectedCompletionDate
        };

        contract.expectedCompletionDate = expectedCompletionDate;
      }

      if (input.acceptanceDeadline !== undefined) {
        changes.acceptanceDeadline = {
          before: contract.acceptanceDeadline,
          after: acceptanceDeadline
        };

        contract.acceptanceDeadline = acceptanceDeadline;
      }

      await contract.save({ session });

      await writeAuditLog(
        {
          adminId: administratorId,
          action: "contract.updated",
          resourceId: contract._id,
          description: "Contract draft details updated.",
          changes,
          metadata: {
            contractNumber: contract.contractNumber,
            contractVersion: contract.version
          }
        },
        context,
        session
      );

    if (!contract) {

/**
 * Issue a contract and generate its secure client access token.
 */
export async function issueContract(
  contractId: string,
  adminId: string,
  context: AuditContext = {}
) {
  const id = validateObjectId(
    contractId,
    "contract ID"
  );

  const administratorId = validateObjectId(
    adminId,
    "administrator ID"
  );

  const session = await mongoose.startSession();

  let issuedContractId: Types.ObjectId | null = null;
  let rawToken: string | null = null;

  try {
    await session.withTransaction(async () => {
      const contract = await Contract.findById(id)
        .session(session);

      if (!contract) {
        throw new ApiError(404, "Contract not found.");
      }

      if (contract.status !== ContractStatus.DRAFT) {
        throw new ApiError(
          409,
          "Only draft contracts can be issued."
        );
      }

      if (
        !contract.acceptanceDeadline ||
        contract.acceptanceDeadline <= new Date()
      ) {
        throw new ApiError(
          400,
          "A valid future acceptance deadline is required before issuing the contract."
        );
      }

      await ensureLatestContractVersion(contract, session);

      const token = randomBytes(32).toString("hex");

      const expiresAt = new Date(
        Math.min(
          contract.acceptanceDeadline.getTime(),
          Date.now() + ACCESS_TOKEN_DURATION_MS
        )
      );

      const accessToken = new ContractAccessToken({
        contractId: contract._id,
        contractVersion: contract.version,
        tokenHash: hashToken(token),
        expiresAt
      });

      await accessToken.save({ session });

      contract.status = ContractStatus.ISSUED;
      contract.issuedAt = new Date();

      await contract.save({ session });

      await writeAuditLog(
        {
          adminId: administratorId,
          action: "contract.issued",
          resourceId: contract._id,
          description: "Contract issued to the client.",
          metadata: {
            contractNumber: contract.contractNumber,
            contractVersion: contract.version,
            accessTokenId: accessToken._id.toString(),
            tokenExpiresAt: expiresAt.toISOString()
          }
        },
        context,
        session
      );

      issuedContractId = contract._id;
      rawToken = token;
    });

    if (!issuedContractId || !rawToken) {
      throw new ApiError(
        500,
        "Contract issuance could not be completed."
      );
    }

    const contract = await Contract.findById(issuedContractId);

    if (!contract) {
      throw new ApiError(
        500,
        "Issued contract could not be retrieved."
      );
    }

    return {
      contract,
      accessToken: rawToken
    };
  } finally {
    await session.endSession();
  }
}

/**
 * Create a new draft revision from an existing contract version.
 *
 * Previous versions remain separate historical documents.
 * Outstanding invitations for the previous version are revoked.
 */
export async function createContractRevision(
  contractId: string,
  adminId: string,
  context: AuditContext = {}
) {
  const id = validateObjectId(
    contractId,
    "contract ID"
  );

  const administratorId = validateObjectId(
    adminId,
    "administrator ID"
  );

  const session = await mongoose.startSession();

  let newContractId: Types.ObjectId | null = null;

  try {
    await session.withTransaction(async () => {
      const previousContract = await Contract.findById(id)
        .session(session);

      if (!previousContract) {
        throw new ApiError(404, "Contract not found.");
      }

      if (previousContract.status === ContractStatus.DRAFT) {
        throw new ApiError(
          409,
          "A revision cannot be created from a draft contract."
        );
      }

      if (previousContract.status === ContractStatus.WITHDRAWN) {
        throw new ApiError(
          409,
          "A revision cannot be created from a withdrawn contract."
        );
      }

      await ensureLatestContractVersion(
        previousContract,
        session
      );

      const revokedAt = new Date();

      await ContractAccessToken.updateMany(
        {
          contractId: previousContract._id,
          contractVersion: previousContract.version,
          revokedAt: null
        },
        {
          $set: { revokedAt }
        },
        { session }
      );

      const revision = new Contract({
        contractNumber: previousContract.contractNumber,

        clientId: previousContract.clientId,
        clientSnapshot: previousContract.clientSnapshot,

        serviceRequestId: previousContract.serviceRequestId,

        quotationId: previousContract.quotationId,
        quotationNumber: previousContract.quotationNumber,

        projectId: previousContract.projectId,

        version: previousContract.version + 1,
        previousVersionId: previousContract._id,

        title: previousContract.title,
        scope: previousContract.scope,

        items: previousContract.items.map((item) => ({
          serviceName: item.serviceName,
          description: item.description,
          deliverables: item.deliverables,
          quantity: item.quantity,
          unitAmountKobo: item.unitAmountKobo,
          totalAmountKobo: item.totalAmountKobo
        })),

        deliverables: [...previousContract.deliverables],

        contractAmountKobo: previousContract.contractAmountKobo,

        terms: [...previousContract.terms],

        startDate: previousContract.startDate,
        expectedCompletionDate: previousContract.expectedCompletionDate,

        acceptanceDeadline: null,

        status: ContractStatus.DRAFT,

        issuedAt: null
      });

      await revision.save({ session });

      await writeAuditLog(
        {
          adminId: administratorId,
          action: "contract.revised",
          resourceId: revision._id,
          description: "New contract revision created.",
          metadata: {
            contractNumber: revision.contractNumber,
            previousVersion: previousContract.version,
            newVersion: revision.version,
            previousContractId: previousContract._id.toString()
          }
        },
        context,
        session
      );

      newContractId = revision._id;
    });

    if (!newContractId) {
      throw new ApiError(
        500,
        "Contract revision could not be created."
      );
    }

    const revision = await Contract.findById(newContractId);

    if (!revision) {
      throw new ApiError(
        500,
        "Created contract revision could not be retrieved."
      );
    }

    return revision;
  } finally {
    await session.endSession();
  }
}

/**
 * Record a client's acceptance or rejection of an issued contract.
 *
 * On acceptance:
 * - Create the acceptance evidence.
 * - Mark the exact contract version as accepted.
 * - Create or update its associated project.
 * - Revoke the contract access tokens.
 * - Record contract and project audit logs.
 *
 * All operations execute within one MongoDB transaction.
 */
export async function respondToContract(
  token: string,
  input: ContractResponseInput,
  context: AuditContext = {}
) {
  const session = await mongoose.startSession();

  let respondedContractId: Types.ObjectId | null = null;

  try {
    await session.withTransaction(async () => {
      const accessToken = await findValidAccessToken(
        token,
        session
      );

      const contract = await Contract.findOne({
        _id: accessToken.contractId,
        version: accessToken.contractVersion
      }).session(session);

      if (
        !contract ||
        contract.status !== ContractStatus.ISSUED
      ) {
        throw new ApiError(
          404,
          "Contract link is invalid or unavailable."
        );
      }

      await ensureLatestContractVersion(contract, session);

      if (
        !contract.acceptanceDeadline ||
        contract.acceptanceDeadline <= new Date()
      ) {
        throw new ApiError(
          410,
          "This contract has expired."
        );
      }

      const respondedAt = new Date();

      if (input.response === "accepted") {
        const acceptedBy = input.acceptedBy?.trim();

        if (!acceptedBy) {
          throw new ApiError(
            400,
            "The name of the person accepting the contract is required."
          );
        }

        if (input.agreementConfirmed !== true) {
          throw new ApiError(
            400,
            "You must explicitly confirm your agreement to the contract."
          );
        }

        const acceptedDocumentFileId =
          input.acceptedDocumentFileId
            ? validateObjectId(
                input.acceptedDocumentFileId,
                "accepted contract document ID"
              )
            : null;

        await ContractAcceptance.create(
          [
            {
              contractId: contract._id,
              contractVersion: contract.version,
              acceptedBy,
              acceptedAt: respondedAt,
              agreementConfirmed: true,
              acceptedDocumentFileId,
              evidence: {
                ipAddress: context.ipAddress ?? null,
                userAgent: context.userAgent ?? null
              }
            }
          ],
          { session }
        );

        contract.status = ContractStatus.ACCEPTED;

        await contract.save({ session });

        // Automatically create the first project or update the
        // existing project when a subsequent revision is accepted.
        // This uses the same transaction session.
        await createOrUpdateProjectFromAcceptedContract(
          contract._id,
          session,
          context
        );
      } else {
        contract.status = ContractStatus.REJECTED;

        await contract.save({ session });
      }

      await ContractAccessToken.updateMany(
        {
          contractId: contract._id,
          contractVersion: contract.version,
          revokedAt: null
        },
        {
          $set: {
            revokedAt: respondedAt
          }
        },
        { session }
      );

      await writeAuditLog(
        {
          adminId: null,
          action: `contract.client_${input.response}`,
          resourceId: contract._id,
          description:
            input.response === "accepted"
              ? "Client accepted the contract."
              : "Client rejected the contract.",
          metadata: {
            contractNumber: contract.contractNumber,
            contractVersion: contract.version
          }
        },
        context,
        session
      );

      respondedContractId = contract._id;
    });

    if (!respondedContractId) {
      throw new ApiError(
        500,
        "Contract response could not be recorded."
      );
    }

    const contract = await Contract.findById(respondedContractId);

    if (!contract) {
      throw new ApiError(
        500,
        "Contract response could not be retrieved."
      );
    }

    return contract;
  } finally {
    await session.endSession();
  }
}

async function validateAcceptedContract(
  contractId: Types.ObjectId,
  session: ClientSession
) {
  const contract = await Contract.findById(contractId).session(session);

  if (!contract) {
    throw new ApiError(404, "Contract not found.");
  }

  if (contract.status !== ContractStatus.ACCEPTED) {
    throw new ApiError(
      409,
      "Only accepted contracts can generate or update projects."
    );
  }

  const newerVersion = await Contract.exists({
    contractNumber: contract.contractNumber,
    version: { $gt: contract.version }
  }).session(session);

  if (newerVersion) {
    throw new ApiError(
      409,
      "Only the latest contract version can be used to update a project."
    );
  }

  const acceptance = await ContractAcceptance.findOne({
    contractId: contract._id,
    contractVersion: contract.version,
    agreementConfirmed: true
  }).session(session);

  if (!acceptance) {
    throw new ApiError(
      409,
      "Valid contract acceptance evidence was not found."
    );
  }

  return contract;
}

async function getContractRelatedRecords(
  contract: Awaited<ReturnType<typeof validateAcceptedContract>>,
  session: ClientSession
) {
  const quotation = await Quotation.findById(contract.quotationId)
    .session(session);

  if (!quotation) {
    throw new ApiError(404, "Associated quotation not found.");
  }

  if (
    quotation.clientId.toString() !== contract.clientId.toString()
  ) {
    throw new ApiError(
      409,
      "Contract and quotation client references do not match."
    );
  }

  const serviceRequest = await ServiceRequest.findById(
    quotation.serviceRequestId
  ).session(session);

  if (!serviceRequest) {
    throw new ApiError(404, "Associated service request not found.");
  }

  if (
    serviceRequest.clientId.toString() !== contract.clientId.toString() ||
    serviceRequest.clientId.toString() !== quotation.clientId.toString()
  ) {
    throw new ApiError(
      409,
      "Contract, quotation and service request client references do not match."
    );
  }

  return {
    quotation,
    serviceRequest
  };
}

async function writeProjectAuditLog(
  projectId: Types.ObjectId,
  contractId: Types.ObjectId,
  contractVersion: number,
  quotationId: Types.ObjectId,
  serviceRequestId: Types.ObjectId,
  action: "project.created" | "project.updated_from_contract_revision",
  description: string,
  session: ClientSession,

): Promise<void> {
  await AuditLog.create(

  );
}

/**
 * Creates a project for the first accepted contract version,
 * or updates the existing project when a subsequent revision
 * is accepted.
 *
 * This function must be called within an existing MongoDB
 * transaction. It does not create, commit or end a session.
 */
export async function createOrUpdateProjectFromAcceptedContract(
  contractId: Types.ObjectId,
  session: ClientSession,
  context: ProjectAuditContext = {}
) {
  const contract = await validateAcceptedContract(
    contractId,
    session
  );

  const {
    quotation,
    serviceRequest
  } = await getContractRelatedRecords(contract, session);

  const serviceNames = serviceRequest.items.map(
    (item) => item.serviceName
  );

  const sourceSnapshot =
    contract.clientSnapshot ?? serviceRequest.clientSnapshot;

  if (!sourceSnapshot) {
    throw new ApiError(
      409,
      "A client snapshot is required to create or update the project."
    );
  }

  const clientSnapshot = {
    fullName: sourceSnapshot.fullName,
    email: sourceSnapshot.email ?? null,
    phone: sourceSnapshot.phone
  };

  let project = contract.projectId
    ? await Project.findById(contract.projectId).session(session)
    : await Project.findOne({
        contractId: contract._id
      }).session(session);

  const isNewProject = !project;

  if (!project) {
    project = new Project({
      projectReference: generateProjectReference(),

      clientId: contract.clientId,
      clientSnapshot,

      serviceRequestId: serviceRequest._id,
      quotationId: quotation._id,
      contractId: contract._id,

      serviceNames,

      title: contract.title,
      description: contract.scope,
      contractAmountKobo: contract.contractAmountKobo,

      status: ProjectStatus.NOT_STARTED,
      startDate: contract.startDate ?? null,
      expectedCompletionDate:
        contract.expectedCompletionDate ?? null
    });
  } else {
    if (
      project.clientId.toString() !== contract.clientId.toString() ||
      project.quotationId.toString() !== quotation._id.toString() ||
      project.serviceRequestId.toString() !== serviceRequest._id.toString()
    ) {
      throw new ApiError(
        409,
        "The existing project references do not match the accepted contract."
      );
    }

    project.contractId = contract._id;
    project.clientSnapshot = clientSnapshot;
    project.serviceNames = serviceNames;

    project.title = contract.title;
    project.description = contract.scope;
    project.contractAmountKobo = contract.contractAmountKobo;

    project.startDate = contract.startDate ?? null;
    project.expectedCompletionDate =
      contract.expectedCompletionDate ?? null;
  }

  await project.save({ session });

  if (!contract.projectId) {
    contract.projectId = project._id;
    await contract.save({ session });
  }

  await writeProjectAuditLog(
    project._id,
    contract._id,
    contract.version,
    quotation._id,
    serviceRequest._id,
    isNewProject
      ? "project.created"
      : "project.updated_from_contract_revision",
    isNewProject
      ? "Project created automatically following contract acceptance."
      : "Project updated following acceptance of a contract revision.",
    session,
    context
  );

  return project;
}

/**
 * Retrieve projects with pagination, search, filtering and sorting.
 *
 * Search fields:
 * - Project reference
 * - Project title
 * - Client name
 *
 * Date filtering applies to project creation dates.
 *
 * Returns mapped project list items and pagination metadata.
 */
export async function getProjects(
  query: ListProjectsQuery
): Promise<ProjectListResponse> {
  const {
    page,
    limit,
    search,
    status,
    clientId,
    serviceRequestId,
    quotationId,
    contractId,
    from,
    to,
    sortBy,
    sortOrder
  } = query;

  const filter: ProjectListFilter = {
    isDeleted: false
  };

  if (status) {
    filter.status = status;
  }

  if (clientId) {
    filter.clientId = validateObjectId(
      clientId,
      "client ID"
    );
  }

  if (serviceRequestId) {
    filter.serviceRequestId = validateObjectId(
      serviceRequestId,
      "service request ID"
    );
  }

  if (quotationId) {
    filter.quotationId = validateObjectId(
      quotationId,
      "quotation ID"
    );
  }

  if (contractId) {
    filter.contractId = validateObjectId(
      contractId,
      "contract ID"
    );
  }

  if (from || to) {
    filter.createdAt = {
      ...(from ? { $gte: new Date(from) } : {}),
      ...(to ? { $lte: new Date(to) } : {})
    };
  }

  if (search) {
    const searchPattern = new RegExp(
      escapeRegExp(search),
      "i"
    );

    filter.$or = [
      { projectReference: searchPattern },
      { title: searchPattern },
      { "clientSnapshot.fullName": searchPattern }
    ];
  }

  const skip = (page - 1) * limit;

  const sortDirection: 1 | -1 =
    sortOrder === "asc" ? 1 : -1;

  const sortOptions: Record<string, 1 | -1> = {
    [sortBy]: sortDirection
  };

  const [
    projects,
    total
  ] = await Promise.all([
    Project.find(filter)
      .sort(sortOptions)
      .skip(skip)
      .limit(limit)
      .lean(),

    Project.countDocuments(filter)
  ]);

  const totalPages = Math.ceil(total / limit);

  return {
    projects: projects.map((project) =>
      mapProjectListItem(
        project as unknown as ProjectWithId
      )
    ),

    pagination: {
      page,
      limit,
      total,
      totalPages,
      hasNextPage: page < totalPages,
      hasPreviousPage: page > 1
    }
  };
}

import { Types } from "mongoose";

import type {
  IQuotation,
  IQuotationItem,
  IClientSummary
} from "@primeace/shared";

import type { QuotationDocument } from "./Quotation.js";

type QuotationWithIdentifiers = QuotationDocument & {
  _id: Types.ObjectId;
  items: Array<
    QuotationDocument["items"][number] & {
      _id: Types.ObjectId;
    }
  >;
};

function serializeId(
  value: Types.ObjectId | string | null | undefined
): string | null {
  return value == null ? null : value.toString();
}

function serializeQuotationItem(
  item: QuotationWithIdentifiers["items"][number]
): IQuotationItem {
  return {
    id: item._id.toString(),
    serviceRequestItemId: serializeId(
      item.serviceRequestItemId
    ),
    serviceName: item.serviceName,
    description: item.description,
    deliverables: item.deliverables,
    quantity: item.quantity,
    unitAmountKobo: item.unitAmountKobo,
    totalAmountKobo: item.totalAmountKobo
  };
}

export function serializeQuotation(
  quotation: QuotationWithIdentifiers,
  client: IClientSummary
): IQuotation {
  return {
    id: quotation._id.toString(),
    quotationNumber: quotation.quotationNumber,

    clientId: quotation.clientId.toString(),
    client,

    serviceRequestId: quotation.serviceRequestId.toString(),
    serviceRequestReference:
      quotation.serviceRequestReference,

    version: quotation.version,
    previousVersionId: serializeId(
      quotation.previousVersionId
    ),

    items: quotation.items.map(serializeQuotationItem),

    subtotalKobo: quotation.subtotalKobo,
    discountKobo: quotation.discountKobo,
    totalAmountKobo: quotation.totalAmountKobo,

function createQuotationNumber(): string {

function calculateQuotationAmounts(
  items: QuotationItemInput[],
  discountKobo: number
) {
  if (items.length === 0) {
    throw new ApiError(
      400,
      "A quotation must contain at least one item."
    );
  }

  if (
    !Number.isSafeInteger(discountKobo) ||
    discountKobo < 0
  ) {
    throw new ApiError(400, "Invalid quotation discount.");
  }

  const calculatedItems = items.map((item) => {
    if (
      !Number.isSafeInteger(item.quantity) ||
      item.quantity < 1 ||
      !Number.isSafeInteger(item.unitAmountKobo) ||
      item.unitAmountKobo < 0
    ) {
      throw new ApiError(
        400,
        "Invalid quotation item amount or quantity."
      );
    }

    const totalAmountKobo =
      item.quantity * item.unitAmountKobo;

    if (!Number.isSafeInteger(totalAmountKobo)) {
      throw new ApiError(
        400,
        "Quotation item total exceeds the supported amount."
      );
    }

    return {
      serviceRequestItemId: item.serviceRequestItemId
        ? validateObjectId(
            item.serviceRequestItemId,
            "service request item ID"
          )
        : null,

      serviceName: item.serviceName.trim(),
      description: item.description.trim(),
      deliverables: item.deliverables.map(
        (value) => value.trim()
      ),
      quantity: item.quantity,
      unitAmountKobo: item.unitAmountKobo,
      totalAmountKobo
    };
  });

  const subtotalKobo = calculatedItems.reduce(
    (total, item) => total + item.totalAmountKobo,
    0
  );

  if (!Number.isSafeInteger(subtotalKobo)) {
    throw new ApiError(
      400,
      "Quotation subtotal exceeds the supported amount."
    );
  }

  if (discountKobo > subtotalKobo) {
    throw new ApiError(
      400,
      "Quotation discount cannot exceed the subtotal."
    );
  }

  return {
    items: calculatedItems,
    subtotalKobo,
    discountKobo,
    totalAmountKobo: subtotalKobo - discountKobo
  };
}

async function validateQuotationItems(
  serviceRequestId: Types.ObjectId,
  items: QuotationItemInput[],
  session: mongoose.ClientSession
) {
  const serviceRequest = await ServiceRequest.findOne({
    _id: serviceRequestId,
    isDeleted: false
  }).session(session);

  if (!serviceRequest) {
    throw new ApiError(404, "Service request not found.");

    serviceRequest.items.map(
      (item) =>

            $match: {
              isDeleted: false
            }
          },
          {
            $project: {
              _id: 1
            }
          }
        ],
        as: "activeClient"
      }
    },
    {
      $lookup: {
        from: ServiceRequest.collection.name,
        localField: "serviceRequestId",
        foreignField: "_id",
        pipeline: [
          {
            $match: {
              isDeleted: false
            }
          },
          {
            $project: {
              _id: 1
            }
          }
        ],
        as: "activeServiceRequest"
      }
    },
    {
      $match: {
        activeClient: { $ne: [] },
        activeServiceRequest: { $ne: [] }
      }
    },
    {
      $facet: {
        quotations: [
          {
            $sort: sort
          },
          {
            $skip: skip
          },
          {
            $limit: limit
          },
          {
            $project: {
              activeClient: 0,
              activeServiceRequest: 0
            }
          }
        ],
        total: [
          {
            $count: "count"
          }
        ]
      }
    }
  ];

  const [result] = await Quotation.aggregate<QuotationListAggregationResult>(
    pipeline
  );

  const quotations = result?.quotations ?? [];
  const total = result?.total[0]?.count ?? 0;

  return {
    quotations,
    page,
    limit,
    total,
    totalPages: Math.ceil(total / limit)
  };
}

 * Record a client's response to an issued quotation.

export async function respondToQuotation(
  token: string,
  input: ClientResponseInput,
  context: AuditContext = {}
) {
  const session = await mongoose.startSession();

  let quotationId: Types.ObjectId | null = null;

  try {
    await session.withTransaction(async () => {
      const accessToken = await findValidAccessToken(
        token,
        session
      );

      const quotation = await Quotation.findOne({
        _id: accessToken.quotationId,
        quotationNumber: accessToken.quotationNumber,
        version: accessToken.quotationVersion
      }).session(session);

      if (
        !quotation ||
        quotation.status !== QuotationStatus.SENT
      ) {
        throw new ApiError(
          404,
          "Quotation link is invalid or unavailable."
        );
      }

      if (quotation.validUntil <= new Date()) {
        throw new ApiError(
          410,
          "This quotation has expired."
        );
      }

      const responseStatus = {
        approved: QuotationStatus.APPROVED,
        declined: QuotationStatus.DECLINED,
        revision_requested: QuotationStatus.REVISION_REQUESTED
      } as const;

      const respondedAt = new Date();

      const [quotationResponse] = await QuotationResponse.create(
        [
          {
            quotationId: quotation._id,
            quotationNumber: quotation.quotationNumber,
            quotationVersion: quotation.version,
            response: input.response,
            comments: input.comments ?? null,
            accessTokenId: accessToken._id,
            respondedAt
          }
        ],
        { session }
      );

      quotation.status = responseStatus[input.response];
      quotation.respondedAt = respondedAt;

      await quotation.save({ session });

      await QuotationAccessToken.updateMany(
        {
          quotationId: quotation._id,
          quotationVersion: quotation.version,
          revokedAt: null
        },
        {
          $set: { revokedAt: respondedAt }
        },
        { session }
      );

      await writeAuditLog(
        {
          adminId: null,
          action: `quotation.client_${input.response}`,
          resourceId: quotation._id,
          description: `Client ${input.response.replaceAll("_", " ")} the quotation.`,
          metadata: {
            quotationNumber: quotation.quotationNumber,
            quotationVersion: quotation.version,
            responseId: quotationResponse?._id.toString() ?? null
          }
        },
        context,
        session
      );

      quotationId = quotation._id;
    });

    if (!quotationId) {
      throw new ApiError(
        500,
        "Quotation response could not be recorded."
      );
    }

    return Quotation.findById(quotationId);
  } finally {
    await session.endSession();
  }
}

export async function createAdminReview(
  input: CreateAdminReviewInput,
  adminId: string,
  context: AuditContext = {}
) {
  const projectId = validateObjectId(input.projectId, "project ID");
  const actorId = validateObjectId(adminId, "administrator ID");

            consentToPublish: true,
            submissionMethod: "admin",
            submittedBy: actorId,

        );
      }

      await writeAuditLog(

  } finally {
    await session.endSession();
  }

  if (!createdReview) {
    throw new ApiError(
      500,
      "The review could not be created."
    );
  }

  return createdReview;

export async function getReviews(
  input: ListReviewsInput = {}

  const page = input.page ?? 1;
  const limit = input.limit ?? 20;

  if (input.status) {
    filter.status = input.status;
  }

  if (input.submissionMethod) {
    filter.submissionMethod = input.submissionMethod;
  }

  if (input.search?.trim()) {
    const searchRegex = new RegExp(
      escapeRegex(input.search.trim()),
      "i"
    );

    filter.$or = [
      { comment: searchRegex },
      { displayName: searchRegex }
    ];
  }

  const [reviews, total] = await Promise.all([
    Review.find(filter)
      .populate(
        "projectId",
        "projectReference title status"
      )
      .populate(
        "clientId",
        "fullName email phone"
      )
      .populate(
        "submittedBy",
        "fullName email"
      )
      .populate(
        "moderatedBy",
        "fullName email"
      )
      .sort({ submittedAt: -1 })
      .skip((page - 1) * limit)
      .limit(limit)
      .lean(),

    Review.countDocuments(filter)
  ]);

  return {
    reviews,
    pagination: {
      page,
      limit,
      total,
      totalPages: Math.ceil(total / limit)
    }
  };
}

/**
 * Moderates a review and optionally changes its featured status.
 */
export async function moderateReview(
  reviewId: string,
  input: ModerateReviewInput,
  adminId: string,
  context: AuditContext = {}
) {
  const id = validateObjectId(reviewId, "review ID");
  const actorId = validateObjectId(adminId, "administrator ID");

  const session = await mongoose.startSession();

  let moderatedReview: InstanceType<typeof Review> | null = null;

  try {
    await session.withTransaction(async () => {
      const review = await Review.findById(id).session(session);

      if (!review) {
        throw new ApiError(404, "Review not found.");
      }

      const previousStatus = review.status;
      const previousFeatured = review.isFeatured;

      if (
        input.status === ReviewStatus.APPROVED &&
        !review.consentToPublish
      ) {
        throw new ApiError(
          400,
          "This review cannot be approved without explicit publication consent."
        );
      }

      if (
        input.isFeatured === true &&
        (
          input.status !== ReviewStatus.APPROVED ||
          !review.consentToPublish
        )
      ) {
        throw new ApiError(
          400,
          "Only approved reviews with publication consent can be featured."
        );
      }

      review.status = input.status;

      if (input.isFeatured !== undefined) {
        review.isFeatured = input.isFeatured;
      }

      if (input.status !== ReviewStatus.APPROVED) {
        review.isFeatured = false;
      }

      if (input.moderationNote !== undefined) {
        review.moderationNote =
          input.moderationNote?.trim() || null;
      }

      review.moderatedAt = new Date();
      review.moderatedBy = actorId;

      await review.save({ session });

      await writeAuditLog(
        {
          actorId,
          action: "review.moderated",
          resourceId: review._id,
          description: "An administrator moderated a client review.",
          changes: {
            previousStatus,
            newStatus: review.status,
            previousFeatured,
            newFeatured: review.isFeatured,
            moderationNote: review.moderationNote
          },
          context
        },
        session
      );

      moderatedReview = review;
    });
  } finally {
    await session.endSession();
  }

  if (!moderatedReview) {
    throw new ApiError(
      500,
      "The review could not be moderated."
    );
  }

  return moderatedReview;
}

import mongoose from "mongoose";

import { RequestStatus } from "@primeace/shared";

import { ApiError } from "../../shared/errors/ApiError.js";

type Answers = Record<string, unknown>;

interface ServiceFormOption {
  value: string;
}

interface ServiceFormCondition {
  fieldKey: string;
  operator: string;
  value: unknown;
}

interface ServiceFormField {
  key: string;
  label: string;
  type: string;
  required: boolean;
  options?: ServiceFormOption[] | null;
  conditions?: ServiceFormCondition[] | null;
  validation?: {
    minLength?: number | null;
    maxLength?: number | null;
    min?: number | null;
    max?: number | null;
    pattern?: string | null;

}

interface FormField extends ServiceFormField {}

interface ServiceRequestResult {

function isEmpty(value: unknown): boolean {
  return (
    value === undefined ||
    value === null ||
    value === "" ||
    (Array.isArray(value) && value.length === 0)
  );
}

function conditionMatches(
  condition: ServiceFormCondition,
  answers: Answers
): boolean {
  const answer = answers[condition.fieldKey];
  const expected = condition.value;

  switch (condition.operator) {
    case "equals":
      return answer === expected;

    case "not_equals":
      return answer !== expected;

    case "contains":
      return (
        (Array.isArray(answer) &&
          answer.some((item) => Object.is(item, expected))) ||
        (typeof answer === "string" &&
          typeof expected === "string" &&
          answer.includes(expected))
      );

    case "not_contains":
      return !(
        (Array.isArray(answer) &&
          answer.some((item) => Object.is(item, expected))) ||
        (typeof answer === "string" &&
          typeof expected === "string" &&
          answer.includes(expected))
      );

    default:
      return false;
  }
}

function isFieldVisible(
  field: FormField,
  answers: Answers
): boolean {
  return (field.conditions ?? []).every((condition) =>
    conditionMatches(condition, answers)
  );
}

function isValidEmail(value: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value);
}

function isValidDate(value: string): boolean {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    return false;
  }

  const timestamp = Date.parse(`${value}T00:00:00.000Z`);

  if (Number.isNaN(timestamp)) {
    return false;
  }

  return (
    new Date(timestamp).toISOString().slice(0, 10) === value
  );
}

function validateFieldAnswer(
  field: FormField,
  value: unknown
): void {
  if (isEmpty(value)) {
    if (field.required) {
      throw new ApiError(400, `${field.label} is required.`);
    }

    return;
  }

  const validation = field.validation;

  const optionValues = (field.options ?? []).map(
    (option) => option.value
  );

  switch (field.type) {
    case "text":
    case "textarea":
    case "email":
    case "tel":
    case "date": {
      if (typeof value !== "string") {
        throw new ApiError(
          400,
          `${field.label} must be a text value.`
        );
      }

      if (field.type === "email" && !isValidEmail(value)) {
        throw new ApiError(
          400,
          `${field.label} must be a valid email address.`
        );
      }

      if (field.type === "date" && !isValidDate(value)) {
        throw new ApiError(
          400,
          `${field.label} must be a valid date.`
        );
      }

      if (
        validation?.minLength != null &&
        value.length < validation.minLength
      ) {
        throw new ApiError(
          400,
          `${field.label} must contain at least ${validation.minLength} characters.`
        );
      }

      if (
        validation?.maxLength != null &&
        value.length > validation.maxLength
      ) {
        throw new ApiError(
          400,
          `${field.label} must not exceed ${validation.maxLength} characters.`
        );
      }

      if (validation?.pattern) {
        let pattern: RegExp;

        try {
          pattern = new RegExp(validation.pattern);
        } catch {
          throw new ApiError(
            500,
            `The published validation pattern for "${field.label}" is invalid.`
          );
        }

        if (!pattern.test(value)) {
          throw new ApiError(
            400,
            `${field.label} has an invalid format.`
          );
        }
      }

      return;
    }

    case "number": {
      if (
        typeof value !== "number" ||
        !Number.isFinite(value)
      ) {
        throw new ApiError(
          400,
          `${field.label} must be a valid number.`
        );
      }

      if (
        validation?.min != null &&
        value < validation.min
      ) {
        throw new ApiError(
          400,
          `${field.label} must be at least ${validation.min}.`
        );
      }

      if (
        validation?.max != null &&
        value > validation.max
      ) {
        throw new ApiError(
          400,
          `${field.label} must not exceed ${validation.max}.`
        );
      }

      return;
    }

    case "select":
    case "radio": {
      if (
        typeof value !== "string" ||
        !optionValues.includes(value)
      ) {
        throw new ApiError(
          400,
          `Select a valid option for ${field.label}.`
        );
      }

      return;
    }

    case "multiselect": {
      if (
        !Array.isArray(value) ||
        !value.every(
          (item) =>
            typeof item === "string" &&
            optionValues.includes(item)
        )
      ) {
        throw new ApiError(
          400,
          `Select valid options for ${field.label}.`
        );
      }

      return;
    }

    case "checkbox": {
      if (typeof value !== "boolean") {
        throw new ApiError(
          400,
          `${field.label} must be a valid checkbox value.`
        );
      }

      if (field.required && value !== true) {
        throw new ApiError(
          400,
          `${field.label} must be accepted.`
        );
      }

      return;
    }

    default:
      throw new ApiError(
        400,
        `Unsupported field type for ${field.label}.`
      );
  }
}

function validateFormAnswers(
  fields: FormField[],
  submittedAnswers: Answers
): Answers {
  const answerableFields = fields.filter(
    (field) => field.type !== "file"
  );

  const allowedKeys = new Set(
    answerableFields.map((field) => field.key)
  );

  for (const key of Object.keys(submittedAnswers)) {
    if (!allowedKeys.has(key)) {
      throw new ApiError(
        400,
        `The submitted form contains an unknown field: ${key}.`
      );
    }
  }

  const validatedAnswers: Answers = {};

  for (const field of answerableFields) {
    if (!isFieldVisible(field, submittedAnswers)) {
      continue;
    }

    const value = submittedAnswers[field.key];

    validateFieldAnswer(field, value);

    if (value !== undefined) {
      validatedAnswers[field.key] = value;
    }
  }

  return validatedAnswers;
}

function normalizePhone(phone: string): string {
  return phone.trim().replace(/[^\d+]/g, "");
}

function normalizeFormFields(
  fields: ServiceFormField[]
): FormField[] {
  return fields.map((field: ServiceFormField) => ({
    key: field.key,
    label: field.label,
    type: field.type,
    required: field.required,

    options: (field.options ?? []).map(
      (option: ServiceFormOption) => ({
        value: option.value
      })
    ),

    conditions: (field.conditions ?? []).map(
      (condition: ServiceFormCondition) => ({
        fieldKey: condition.fieldKey,
        operator: condition.operator,
        value: condition.value
      })
    ),

    validation: field.validation
      ? {

export async function createPublicServiceRequest(
  input: CreatePublicServiceRequestInput
): Promise<ServiceRequestResult> {
  const session = await mongoose.startSession();

  try {
    const result = await session.withTransaction<ServiceRequestResult>(
      async () => {
        const requestItems = [];

        for (const item of input.items) {
          const service = await Service.findOne({
            _id: item.serviceId,
            isActive: true,
            isDeleted: false
          }).session(session);

          if (!service) {
            throw new ApiError(
              400,
              "One or more selected services are unavailable."
            );
          }

          const category = await ServiceCategory.findOne({
            _id: service.categoryId,
            isActive: true,
            isDeleted: false
          }).session(session);

          if (!category) {
            throw new ApiError(
              400,
              "One or more selected services are unavailable."
            );
          }

          const form = await ServiceForm.findOne({
            serviceId: service._id,
            isActive: true,
            version: item.formVersion
          }).session(session);

          if (!form) {
            throw new ApiError(
              409,
              `The form for ${service.name} has changed. Please refresh and try again.`
            );
          }

          const fields = normalizeFormFields(
            form.fields as unknown as ServiceFormField[]
          );

          const answers = validateFormAnswers(
            fields,
            item.answers
          );

          requestItems.push({
            serviceId: service._id,
            serviceName: service.name,
            categoryId: category._id,
            categoryName: category.name,
            formVersion: form.version,
            answers,
            fileIds: []
          });
        }

        const email =
          input.client.email?.trim().toLowerCase() || null;

        const normalizedPhone = normalizePhone(
          input.client.phone
        );

        const existingClient = email
          ? await Client.findOne({
              normalizedEmail: email,
              isDeleted: false
            }).session(session)
          : null;

        const client = existingClient ?? new Client({
          fullName: input.client.fullName,
          email,
          normalizedEmail: email,
          phone: input.client.phone.trim(),
          normalizedPhone,
          organisation:
            input.client.organisation?.trim() || null
        });

        if (existingClient) {
          client.fullName = input.client.fullName;
          client.email = email;
          client.phone = input.client.phone.trim();
          client.normalizedPhone = normalizedPhone;
          client.organisation =
            input.client.organisation?.trim() || null;
        }

        await client.save({ session });

        const reference = generateServiceRequestReference();
        const submittedAt = new Date();

        const serviceRequest = new ServiceRequest({
          reference,
          clientId: client._id,
          clientSnapshot: {
            fullName: client.fullName,
            email: client.email ?? null,
            phone: client.phone,
            organisation: client.organisation ?? null
          },
          items: requestItems,
          status: RequestStatus.NEW,
          submittedAt
        });

        await serviceRequest.save({ session });

        return {
          reference,
          submittedAt
        };
      }
    );

    if (!result) {
      throw new ApiError(
        500,
        "The service request could not be completed."
      );
    }

    return result;
  } finally {
    await session.endSession();
  }
}

function buildServiceRequestFilter(
  options: ListServiceRequestsOptions
): Record<string, unknown> {
  const filter: Record<string, unknown> = {
    isDeleted: false
  };

  if (options.status) {
    filter.status = options.status;
  }

  if (options.serviceId) {
    filter["items.serviceId"] = assertObjectId(
      options.serviceId,
      "service ID"
    );
  }

  if (options.categoryId) {
    filter["items.categoryId"] = assertObjectId(
      options.categoryId,
      "category ID"
    );
  }

  if (options.submittedFrom || options.submittedTo) {
    const submittedAt: Record<string, Date> = {};

    if (options.submittedFrom) {
      submittedAt.$gte = new Date(options.submittedFrom);
    }

    if (options.submittedTo) {
      submittedAt.$lte = new Date(options.submittedTo);
    }

    filter.submittedAt = submittedAt;
  }

  if (options.search) {
    const search = new RegExp(
      escapeRegex(options.search),
      "i"
    );

    filter.$or = [
      { reference: search },
      { "clientSnapshot.fullName": search },
      { "clientSnapshot.email": search },
      { "clientSnapshot.phone": search }
    ];
  }

  return filter;
}

export async function listServiceRequests(
  options: ListServiceRequestsOptions
): Promise<{
  requests: ServiceRequestListItem[];
  pagination: {
    page: number;
    limit: number;
    total: number;
    totalPages: number;
  };
}> {
  if (
    options.submittedFrom &&
    options.submittedTo &&
    new Date(options.submittedFrom) > new Date(options.submittedTo)
  ) {
    throw new ApiError(
      422,
      "The submission start date cannot be later than the end date."
    );
  }

  const filter = buildServiceRequestFilter(options);

  const sortDirection = options.sortOrder === "asc" ? 1 : -1;

  const sort: Record<string, 1 | -1> = {
    [options.sortBy]: sortDirection
  };

  const skip = (options.page - 1) * options.limit;

  const [requests, total] = await Promise.all([
    ServiceRequest.find(filter)
      .select(
        "_id reference clientSnapshot items.serviceId items.serviceName items.categoryId items.categoryName status submittedAt createdAt updatedAt"
      )
      .sort(sort)
      .skip(skip)
      .limit(options.limit)
      .lean<ServiceRequestListItem[]>(),

    ServiceRequest.countDocuments(filter)
  ]);

  return {
    requests,
    pagination: {
      page: options.page,

export async function updateServiceRequestStatus(
  options: UpdateServiceRequestStatusOptions
): Promise<ServiceRequestDetails> {
  const requestId = assertObjectId(
    options.requestId,
    "service request ID"
  );

  const adminId = assertObjectId(
    options.adminId,
    "administrator ID"
  );

  const session = await mongoose.startSession();

  try {
    await session.withTransaction(async () => {
      const request = await ServiceRequest.findOne({
        _id: requestId,
        isDeleted: false
      }).session(session);

      if (!request) {
        throw new ApiError(404, "Service request not found.");
      }

      const currentStatus = request.status;

      if (currentStatus === options.status) {
        throw new ApiError(
          422,
          "The service request already has this status."
        );
      }

      const allowedStatuses =
        allowedStatusTransitions[currentStatus];

      if (!allowedStatuses.includes(options.status)) {
        throw new ApiError(
          422,
          `A service request cannot transition from ${currentStatus} to ${options.status}.`
        );
      }

      request.status = options.status;

      await request.save({ session });

      await AuditLog.create(
        [
          {
            actorId: adminId,
            action: "service_request.status_updated",
            resourceType: "service_request",
            resourceId: requestId,
            description: `Service request ${request.reference} status changed from ${currentStatus} to ${options.status}.`,
            changes: {
              before: {
                status: currentStatus
              },
              after: {
                status: options.status
              }
            },
            metadata: null,
            ipAddress: options.auditMetadata.ipAddress,
            userAgent: options.auditMetadata.userAgent
          }
        ],
        { session }
      );
    });
  } finally {
    await session.endSession();
  }

  return getServiceRequestById(options.requestId);
}

import { Types } from "mongoose";

import { ApiError } from "../../shared/errors/ApiError.js";

import { Service } from "./Service.js";
import { ServiceCategory } from "./ServiceCategory.js";
import { ServiceForm } from "./ServiceForm.js";

async function getActiveServiceById(serviceId: string) {
  if (!Types.ObjectId.isValid(serviceId)) {
    throw new ApiError(400, "Invalid service ID.");
  }

  const service = await Service.findOne({
    _id: serviceId,
    isActive: true,
    isDeleted: false
  }).lean();

  if (!service) {
    throw new ApiError(404, "Service not found.");
  }

  const category = await ServiceCategory.findOne({
    _id: service.categoryId,
    isActive: true,
    isDeleted: false
  }).lean();

  if (!category) {
    throw new ApiError(404, "Service not found.");
  }

  return { service, category };
}

export async function getPublicServices() {
  const categories = await ServiceCategory.find({
    isActive: true,
    isDeleted: false
  })
    .sort({ displayOrder: 1, name: 1 })
    .lean();

  if (categories.length === 0) {
    return { categories: [] };
  }

  const services = await Service.find({
    categoryId: { $in: categories.map((category) => category._id) },
    isActive: true,
    isDeleted: false
  })
    .sort({ displayOrder: 1, name: 1 })
    .lean();

  return {
    categories: categories.map((category) => ({
      ...category,
      services: services.filter(
        (service) =>
          service.categoryId.toString() === category._id.toString()
      )
    }))
  };
}

export async function getPublicService(serviceId: string) {
  return getActiveServiceById(serviceId);
}

export async function getPublicServiceForm(serviceId: string) {
  const { service } = await getActiveServiceById(serviceId);

  const form = await ServiceForm.findOne({
    serviceId: service._id,
    isActive: true
  })
    .select("serviceId version fields publishedAt")
    .lean();

  if (!form) {
    throw new ApiError(404, "No published form is available for this service.");
  }

  return {
    serviceId: form.serviceId,
    version: form.version,
    publishedAt: form.publishedAt,
    fields: form.fields
      .slice()
      .sort((a, b) => a.displayOrder - b.displayOrder)
      .map((field) => {
        const {
          acceptedMimeTypes,
          maxFiles,
          maxFileSizeBytes,
          ...publicField
        } = field;

        return publicField;
      })
  };
}

import { createServer } from "node:http";

import app from "./app.js";
import { env } from "./config/env.js";
import {
  connectDatabase,
  disconnectDatabase
} from "./config/database.js";

const server = createServer(app);

const SHUTDOWN_TIMEOUT_MS = 10_000;

let isShuttingDown = false;
let isListening = false;

async function startServer(): Promise<void> {
  await connectDatabase();

  if (isShuttingDown) {
    await disconnectDatabase();
    return;
  }

  await new Promise<void>((resolve, reject) => {
    const handleError = (error: Error): void => {
      server.off("listening", handleListening);
      reject(error);
    };

    const handleListening = (): void => {
      server.off("error", handleError);
      isListening = true;
      resolve();
    };

    server.once("error", handleError);
    server.once("listening", handleListening);

    server.listen(env.PORT);
  });

  if (isShuttingDown) {
    await shutdown("STARTUP");
    return;
  }

  console.log(`PrimeAce API listening on port ${env.PORT}.`);
  console.log(`Environment: ${env.NODE_ENV}`);
}

async function shutdown(signal: string): Promise<void> {
  if (isShuttingDown) {
    return;
  }

  isShuttingDown = true;

  console.log(`\n${signal} received. Shutting down gracefully...`);

  const forceCloseTimer = setTimeout(() => {
    console.error("Graceful shutdown timed out. Closing connections.");

    server.closeAllConnections();
  }, SHUTDOWN_TIMEOUT_MS);

  forceCloseTimer.unref();

  try {
    if (isListening) {
      await new Promise<void>((resolve, reject) => {
        server.close((error) => {
          if (error) {
            reject(error);
            return;
          }

          resolve();
        });
      });
    }

    await disconnectDatabase();

    console.log("PrimeAce API shut down successfully.");
    process.exitCode = 0;
  } catch (error) {
    console.error("Error during server shutdown:", error);
    process.exitCode = 1;
  } finally {
    clearTimeout(forceCloseTimer);
  }
}

process.on("SIGINT", () => {
  void shutdown("SIGINT");
});

process.on("SIGTERM", () => {
  void shutdown("SIGTERM");
});

startServer().catch(async (error: unknown) => {
  console.error("Failed to start PrimeAce API:", error);

  try {
    await disconnectDatabase();
  } catch (disconnectError) {
    console.error("Database cleanup failed:", disconnectError);
  }

  process.exitCode = 1;
});

  const isProduction = env.NODE_ENV === "production";

  return {
    secure: isProduction,
    sameSite: "lax",
    path: "/"
  };
}

export function getAdminSessionCookieOptions(): CookieOptions {
  return {
    ...getCookieSecurityOptions(),
    httpOnly: true,
    maxAge: ADMIN_SESSION_DURATION_MS
  };
}

export function getAdminCsrfCookieOptions(): CookieOptions {
  return {
    ...getCookieSecurityOptions(),
    httpOnly: false,
    maxAge: ADMIN_SESSION_DURATION_MS
  };
}

export function getExpiredCookieOptions(): CookieOptions {
  return {
    ...getCookieSecurityOptions(),
    httpOnly: true
  };
}

export function getExpiredCsrfCookieOptions(): CookieOptions {
  return {
    ...getCookieSecurityOptions(),
    httpOnly: false
  };
}
