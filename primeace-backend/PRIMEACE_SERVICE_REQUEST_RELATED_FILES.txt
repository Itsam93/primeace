import {
  model,
  Schema,
  Types
} from "mongoose";

import { RequestStatus } from "@primeace/shared";
import { softDeleteFields } from "../../database/softDelete.js";

interface ServiceRequestItem {
  serviceId: Types.ObjectId;
  serviceName: string;
  categoryId: Types.ObjectId;
  categoryName: string;
  formVersion: number;
  answers: Record<string, unknown>;
  fileIds: Types.ObjectId[];
}

interface ServiceRequestRecord {
  reference: string;
  clientId: Types.ObjectId;

  clientSnapshot: {
    fullName: string;
    email: string | null;
    phone: string;
    organisation: string | null;
  };

  items: ServiceRequestItem[];
  status: RequestStatus;
  submittedAt: Date;

  isDeleted: boolean;
  deletedAt: Date | null;
  deletedBy: Types.ObjectId | null;

  createdAt: Date;
  updatedAt: Date;
}

const serviceRequestItemSchema = new Schema<ServiceRequestItem>(
  {
    serviceId: {
      type: Schema.Types.ObjectId,
      ref: "Service",
      required: true
    },

    serviceName: {
      type: String,
      required: true,
      trim: true
    },

    categoryId: {
      type: Schema.Types.ObjectId,
      ref: "ServiceCategory",
      required: true
    },

    categoryName: {
      type: String,
      required: true,
      trim: true
    },

    formVersion: {
      type: Number,
      required: true,
      min: 1
    },

    answers: {
      type: Schema.Types.Mixed,
      required: true,
      default: {}
    },

    fileIds: {
      type: [Schema.Types.ObjectId],
      ref: "File",
      default: []
    }
  },
  {
    timestamps: true
  }
);

const serviceRequestSchema = new Schema<ServiceRequestRecord>(
  {
    reference: {
      type: String,
      required: true,
      unique: true,
      trim: true
    },

    clientId: {
      type: Schema.Types.ObjectId,
      ref: "Client",
      required: true,
      index: true
    },

    clientSnapshot: {
      fullName: {
        type: String,
        required: true,
        trim: true
      },

      email: {
        type: String,
        default: null,
        lowercase: true,
        trim: true
      },

      phone: {
        type: String,
        required: true,
        trim: true
      },

      organisation: {
        type: String,
        default: null,
        trim: true
      }
    },

    items: {
      type: [serviceRequestItemSchema],
      required: true,
      validate: {
        validator: (items: ServiceRequestItem[]) =>
          items.length > 0,
        message: "A service request must contain at least one service."
      }
    },

    status: {
      type: String,
      enum: Object.values(RequestStatus),
      default: RequestStatus.NEW,
      required: true,
      index: true
    },

    submittedAt: {
      type: Date,
      required: true,
      default: Date.now
    },

    ...softDeleteFields
  },
  {
    timestamps: true,
    versionKey: false
  }
);

serviceRequestSchema.index({
  clientId: 1,
  createdAt: -1
});

export const ServiceRequest = model<ServiceRequestRecord>(
  "ServiceRequest",
  serviceRequestSchema
);

interface AuthenticatedAdmin {
  _id: string;
}

function getAdminId(response: Parameters<RequestHandler>[1]): string {
  const admin = response.locals.admin as
    | AuthenticatedAdmin
    | undefined;

  if (!admin?._id) {
    throw new ApiError(401, "Administrator authentication is required.");
  }

  return admin._id;
}

function getAuditMetadata(
  request: Parameters<RequestHandler>[0]
) {
  return {
    ipAddress: request.ip ?? null,
    userAgent: request.get("user-agent") ?? null
  };
}

import mongoose, { Types } from "mongoose";

import { RequestStatus } from "@primeace/shared";

const allowedStatusTransitions: Record<
  RequestStatus,
  readonly RequestStatus[]
> = {
  [RequestStatus.NEW]: [
    RequestStatus.UNDER_REVIEW,
    RequestStatus.DECLINED,
    RequestStatus.CLOSED
  ],

  [RequestStatus.UNDER_REVIEW]: [
    RequestStatus.AWAITING_INFORMATION,
    RequestStatus.QUOTED,
    RequestStatus.DECLINED,
    RequestStatus.CLOSED
  ],

  [RequestStatus.AWAITING_INFORMATION]: [
    RequestStatus.UNDER_REVIEW,
    RequestStatus.DECLINED,
    RequestStatus.CLOSED
  ],

  [RequestStatus.QUOTED]: [
    RequestStatus.CONVERTED,
    RequestStatus.DECLINED,
    RequestStatus.CLOSED
  ],

  [RequestStatus.CONVERTED]: [
    RequestStatus.CLOSED
  ],

  [RequestStatus.DECLINED]: [
    RequestStatus.CLOSED
  ],

  [RequestStatus.CLOSED]: []
};

function escapeRegex(value: string): string {
  return value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
}

function assertObjectId(value: string, field: string): Types.ObjectId {
  if (!Types.ObjectId.isValid(value)) {
    throw new ApiError(400, `Invalid ${field}.`);
  }

  return new Types.ObjectId(value);
}

async function assertServiceRequestExists(
  requestId: Types.ObjectId
): Promise<void> {
  const exists = await ServiceRequest.exists({
    _id: requestId,
    isDeleted: false
  });

  if (!exists) {
    throw new ApiError(404, "Service request not found.");
  }
}

function buildServiceRequestFilter(
  options: ListServiceRequestsOptions
): Record<string, unknown> {
  const filter: Record<string, unknown> = {
    isDeleted: false
  };

  if (options.status) {
    filter.status = options.status;
  }

  if (options.serviceId) {
    filter["items.serviceId"] = assertObjectId(
      options.serviceId,
      "service ID"
    );
  }

  if (options.categoryId) {
    filter["items.categoryId"] = assertObjectId(
      options.categoryId,
      "category ID"
    );
  }

  if (options.submittedFrom || options.submittedTo) {
    const submittedAt: Record<string, Date> = {};

    if (options.submittedFrom) {
      submittedAt.$gte = new Date(options.submittedFrom);
    }

    if (options.submittedTo) {
      submittedAt.$lte = new Date(options.submittedTo);
    }

    filter.submittedAt = submittedAt;
  }

  if (options.search) {
    const search = new RegExp(
      escapeRegex(options.search),
      "i"
    );

    filter.$or = [
      { reference: search },
      { "clientSnapshot.fullName": search },
      { "clientSnapshot.email": search },
      { "clientSnapshot.phone": search }
    ];
  }

  return filter;
}

export async function listServiceRequests(
  options: ListServiceRequestsOptions
): Promise<{
  requests: ServiceRequestListItem[];
  pagination: {
    page: number;
    limit: number;
    total: number;
    totalPages: number;
  };
}> {
  if (
    options.submittedFrom &&
    options.submittedTo &&
    new Date(options.submittedFrom) > new Date(options.submittedTo)
  ) {
    throw new ApiError(
      422,
      "The submission start date cannot be later than the end date."
    );
  }

  const filter = buildServiceRequestFilter(options);

  const sortDirection = options.sortOrder === "asc" ? 1 : -1;

  const sort: Record<string, 1 | -1> = {
    [options.sortBy]: sortDirection
  };

  const skip = (options.page - 1) * options.limit;

  const [requests, total] = await Promise.all([
    ServiceRequest.find(filter)
      .select(
        "_id reference clientSnapshot items.serviceId items.serviceName items.categoryId items.categoryName status submittedAt createdAt updatedAt"
      )
      .sort(sort)
      .skip(skip)
      .limit(options.limit)
      .lean<ServiceRequestListItem[]>(),

    ServiceRequest.countDocuments(filter)
  ]);

  return {
    requests,
    pagination: {
      page: options.page,
      limit: options.limit,
      total,
      totalPages: Math.ceil(total / options.limit)
    }
  };
}

export async function getServiceRequestById(
  requestId: string
): Promise<ServiceRequestDetails> {
  const id = assertObjectId(requestId, "service request ID");

  const request = await ServiceRequest.findOne({
    _id: id,
    isDeleted: false
  })
    .lean<ServiceRequestDetails>();

  if (!request) {
    throw new ApiError(404, "Service request not found.");
  }

  return request;
}

export async function updateServiceRequestStatus(
  options: UpdateServiceRequestStatusOptions
): Promise<ServiceRequestDetails> {
  const requestId = assertObjectId(
    options.requestId,
    "service request ID"
  );

  const adminId = assertObjectId(
    options.adminId,
    "administrator ID"
  );

  const session = await mongoose.startSession();

  try {
    await session.withTransaction(async () => {
      const request = await ServiceRequest.findOne({
        _id: requestId,
        isDeleted: false
      }).session(session);

      if (!request) {
        throw new ApiError(404, "Service request not found.");
      }

      const currentStatus = request.status;

      if (currentStatus === options.status) {
        throw new ApiError(
          422,
          "The service request already has this status."
        );
      }

      const allowedStatuses =
        allowedStatusTransitions[currentStatus];

      if (!allowedStatuses.includes(options.status)) {
        throw new ApiError(
          422,
          `A service request cannot transition from ${currentStatus} to ${options.status}.`
        );
      }

      request.status = options.status;

      await request.save({ session });

      await AuditLog.create(
        [
          {
            actorId: adminId,
            action: "service_request.status_updated",
            resourceType: "service_request",
            resourceId: requestId,
            description: `Service request ${request.reference} status changed from ${currentStatus} to ${options.status}.`,
            changes: {
              before: {
                status: currentStatus
              },
              after: {
                status: options.status
              }
            },
            metadata: null,
            ipAddress: options.auditMetadata.ipAddress,
            userAgent: options.auditMetadata.userAgent
          }
        ],
        { session }
      );
    });
  } finally {
    await session.endSession();
  }

  return getServiceRequestById(options.requestId);
}

export async function createServiceRequestNote(
  options: CreateServiceRequestNoteOptions
): Promise<ServiceRequestNoteItem> {
  const requestId = assertObjectId(
    options.requestId,
    "service request ID"
  );

  const adminId = assertObjectId(
    options.adminId,
    "administrator ID"
  );

  const session = await mongoose.startSession();

  let createdNoteId: Types.ObjectId | null = null;

  try {
    await session.withTransaction(async () => {
      const request = await ServiceRequest.findOne({
        _id: requestId,
        isDeleted: false
      }).session(session);

      if (!request) {
        throw new ApiError(404, "Service request not found.");
      }

      const [note] = await ServiceRequestNote.create(
        [
          {
            serviceRequestId: requestId,
            adminId,
            content: options.content
          }
        ],
        { session }
      );

      if (!note) {
        throw new ApiError(
          500,
          "The internal note could not be created."
        );
      }

      createdNoteId = note._id;

      await AuditLog.create(
        [
          {
            actorId: adminId,
            action: "service_request.note_created",
            resourceType: "service_request",
            resourceId: requestId,
            description: `An internal note was added to service request ${request.reference}.`,
            changes: null,
            metadata: {
              noteId: note._id
            },
            ipAddress: options.auditMetadata.ipAddress,
            userAgent: options.auditMetadata.userAgent
          }
        ],
        { session }
      );
    });
  } finally {
    await session.endSession();
  }

  if (!createdNoteId) {
    throw new ApiError(
      500,
      "The internal note could not be retrieved."
    );
  }

  const createdNote = await ServiceRequestNote.findById(
    createdNoteId
  )
    .populate("adminId", "fullName email")
    .lean<ServiceRequestNoteItem>();

  if (!createdNote) {
    throw new ApiError(
      500,
      "The internal note could not be retrieved."
    );
  }

  return createdNote;
}

export async function listServiceRequestNotes(
  options: ListServiceRequestNotesOptions
): Promise<{
  notes: ServiceRequestNoteItem[];
  pagination: {
    page: number;
    limit: number;
    total: number;
    totalPages: number;
  };
}> {
  const requestId = assertObjectId(
    options.requestId,
    "service request ID"
  );

  await assertServiceRequestExists(requestId);

  const filter = {
    serviceRequestId: requestId
  };

  const skip = (options.page - 1) * options.limit;

  const [notes, total] = await Promise.all([
    ServiceRequestNote.find(filter)
      .populate("adminId", "fullName email")
      .sort({ createdAt: -1 })
      .skip(skip)
      .limit(options.limit)
      .lean<ServiceRequestNoteItem[]>(),

    ServiceRequestNote.countDocuments(filter)
  ]);

  return {
    notes,
    pagination: {
      page: options.page,
      limit: options.limit,
      total,
      totalPages: Math.ceil(total / options.limit)
    }
  };
}

type Answers = Record<string, unknown>;

interface ServiceFormOption {
  value: string;
}

interface ServiceFormCondition {
  fieldKey: string;
  operator: string;
  value: unknown;
}

interface ServiceFormField {
  key: string;
  label: string;
  type: string;
  required: boolean;
  options?: ServiceFormOption[] | null;
  conditions?: ServiceFormCondition[] | null;
  validation?: {
    minLength?: number | null;
    maxLength?: number | null;
    min?: number | null;
    max?: number | null;
    pattern?: string | null;
  } | null;
}

interface FormField extends ServiceFormField {}

interface ServiceRequestResult {
  reference: string;
  submittedAt: Date;
}

function isEmpty(value: unknown): boolean {
  return (
    value === undefined ||
    value === null ||
    value === "" ||
    (Array.isArray(value) && value.length === 0)
  );
}

function conditionMatches(
  condition: ServiceFormCondition,
  answers: Answers
): boolean {
  const answer = answers[condition.fieldKey];
  const expected = condition.value;

  switch (condition.operator) {
    case "equals":
      return answer === expected;

    case "not_equals":
      return answer !== expected;

    case "contains":
      return (
        (Array.isArray(answer) &&
          answer.some((item) => Object.is(item, expected))) ||
        (typeof answer === "string" &&
          typeof expected === "string" &&
          answer.includes(expected))
      );

    case "not_contains":
      return !(
        (Array.isArray(answer) &&
          answer.some((item) => Object.is(item, expected))) ||
        (typeof answer === "string" &&
          typeof expected === "string" &&
          answer.includes(expected))
      );

    default:
      return false;
  }
}

function isFieldVisible(
  field: FormField,
  answers: Answers
): boolean {
  return (field.conditions ?? []).every((condition) =>
    conditionMatches(condition, answers)
  );
}

function isValidEmail(value: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value);
}

function isValidDate(value: string): boolean {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    return false;
  }

  const timestamp = Date.parse(`${value}T00:00:00.000Z`);

  if (Number.isNaN(timestamp)) {
    return false;
  }

  return (
    new Date(timestamp).toISOString().slice(0, 10) === value
  );
}

function validateFieldAnswer(
  field: FormField,
  value: unknown
): void {
  if (isEmpty(value)) {
    if (field.required) {
      throw new ApiError(400, `${field.label} is required.`);
    }

    return;
  }

  const validation = field.validation;

  const optionValues = (field.options ?? []).map(
    (option) => option.value
  );

  switch (field.type) {
    case "text":
    case "textarea":
    case "email":
    case "tel":
    case "date": {
      if (typeof value !== "string") {
        throw new ApiError(
          400,
          `${field.label} must be a text value.`
        );
      }

      if (field.type === "email" && !isValidEmail(value)) {
        throw new ApiError(
          400,
          `${field.label} must be a valid email address.`
        );
      }

      if (field.type === "date" && !isValidDate(value)) {
        throw new ApiError(
          400,
          `${field.label} must be a valid date.`
        );
      }

      if (
        validation?.minLength != null &&
        value.length < validation.minLength
      ) {
        throw new ApiError(
          400,
          `${field.label} must contain at least ${validation.minLength} characters.`
        );
      }

      if (
        validation?.maxLength != null &&
        value.length > validation.maxLength
      ) {
        throw new ApiError(
          400,
          `${field.label} must not exceed ${validation.maxLength} characters.`
        );
      }

      if (validation?.pattern) {
        let pattern: RegExp;

        try {
          pattern = new RegExp(validation.pattern);
        } catch {
          throw new ApiError(
            500,
            `The published validation pattern for "${field.label}" is invalid.`
          );
        }

        if (!pattern.test(value)) {
          throw new ApiError(
            400,
            `${field.label} has an invalid format.`
          );
        }
      }

      return;
    }

    case "number": {
      if (
        typeof value !== "number" ||
        !Number.isFinite(value)
      ) {
        throw new ApiError(
          400,
          `${field.label} must be a valid number.`
        );
      }

      if (
        validation?.min != null &&
        value < validation.min
      ) {
        throw new ApiError(
          400,
          `${field.label} must be at least ${validation.min}.`
        );
      }

      if (
        validation?.max != null &&
        value > validation.max
      ) {
        throw new ApiError(
          400,
          `${field.label} must not exceed ${validation.max}.`
        );
      }

      return;
    }

    case "select":
    case "radio": {
      if (
        typeof value !== "string" ||
        !optionValues.includes(value)
      ) {
        throw new ApiError(
          400,
          `Select a valid option for ${field.label}.`
        );
      }

      return;
    }

    case "multiselect": {
      if (
        !Array.isArray(value) ||
        !value.every(
          (item) =>
            typeof item === "string" &&
            optionValues.includes(item)
        )
      ) {
        throw new ApiError(
          400,
          `Select valid options for ${field.label}.`
        );
      }

      return;
    }

    case "checkbox": {
      if (typeof value !== "boolean") {
        throw new ApiError(
          400,
          `${field.label} must be a valid checkbox value.`
        );
      }

      if (field.required && value !== true) {
        throw new ApiError(
          400,
          `${field.label} must be accepted.`
        );
      }

      return;
    }

    default:
      throw new ApiError(
        400,
        `Unsupported field type for ${field.label}.`
      );
  }
}

function validateFormAnswers(
  fields: FormField[],
  submittedAnswers: Answers
): Answers {
  const answerableFields = fields.filter(
    (field) => field.type !== "file"
  );

  const allowedKeys = new Set(
    answerableFields.map((field) => field.key)
  );

  for (const key of Object.keys(submittedAnswers)) {
    if (!allowedKeys.has(key)) {
      throw new ApiError(
        400,
        `The submitted form contains an unknown field: ${key}.`
      );
    }
  }

  const validatedAnswers: Answers = {};

  for (const field of answerableFields) {
    if (!isFieldVisible(field, submittedAnswers)) {
      continue;
    }

    const value = submittedAnswers[field.key];

    validateFieldAnswer(field, value);

    if (value !== undefined) {
      validatedAnswers[field.key] = value;
    }
  }

  return validatedAnswers;
}

function normalizePhone(phone: string): string {
  return phone.trim().replace(/[^\d+]/g, "");
}

function normalizeFormFields(
  fields: ServiceFormField[]
): FormField[] {
  return fields.map((field: ServiceFormField) => ({
    key: field.key,
    label: field.label,
    type: field.type,
    required: field.required,

    options: (field.options ?? []).map(
      (option: ServiceFormOption) => ({
        value: option.value
      })
    ),

    conditions: (field.conditions ?? []).map(
      (condition: ServiceFormCondition) => ({
        fieldKey: condition.fieldKey,
        operator: condition.operator,
        value: condition.value
      })
    ),

    validation: field.validation
      ? {
          minLength: field.validation.minLength ?? null,
          maxLength: field.validation.maxLength ?? null,
          min: field.validation.min ?? null,
          max: field.validation.max ?? null,
          pattern: field.validation.pattern ?? null
        }
      : null
  }));
}

export async function createPublicServiceRequest(
  input: CreatePublicServiceRequestInput
): Promise<ServiceRequestResult> {
  const session = await mongoose.startSession();

  try {
    const result = await session.withTransaction<ServiceRequestResult>(
      async () => {
        const requestItems = [];

        for (const item of input.items) {
          const service = await Service.findOne({
            _id: item.serviceId,
            isActive: true,
            isDeleted: false
          }).session(session);

          if (!service) {
            throw new ApiError(
              400,
              "One or more selected services are unavailable."
            );
          }

          const category = await ServiceCategory.findOne({
            _id: service.categoryId,
            isActive: true,
            isDeleted: false
          }).session(session);

          if (!category) {
            throw new ApiError(
              400,
              "One or more selected services are unavailable."
            );
          }

          const form = await ServiceForm.findOne({
            serviceId: service._id,
            isActive: true,
            version: item.formVersion
          }).session(session);

          if (!form) {
            throw new ApiError(
              409,
              `The form for ${service.name} has changed. Please refresh and try again.`
            );
          }

          const fields = normalizeFormFields(
            form.fields as unknown as ServiceFormField[]
          );

          const answers = validateFormAnswers(
            fields,
            item.answers
          );

          requestItems.push({
            serviceId: service._id,
            serviceName: service.name,
            categoryId: category._id,
            categoryName: category.name,
            formVersion: form.version,
            answers,
            fileIds: []
          });
        }

        const email =
          input.client.email?.trim().toLowerCase() || null;

        const normalizedPhone = normalizePhone(
          input.client.phone
        );

        const existingClient = email
          ? await Client.findOne({
              normalizedEmail: email,
              isDeleted: false
            }).session(session)
          : null;

        const client = existingClient ?? new Client({
          fullName: input.client.fullName,
          email,
          normalizedEmail: email,
          phone: input.client.phone.trim(),
          normalizedPhone,
          organisation:
            input.client.organisation?.trim() || null
        });

        if (existingClient) {
          client.fullName = input.client.fullName;
          client.email = email;
          client.phone = input.client.phone.trim();
          client.normalizedPhone = normalizedPhone;
          client.organisation =
            input.client.organisation?.trim() || null;
        }

        await client.save({ session });

        const reference = generateServiceRequestReference();
        const submittedAt = new Date();

        const serviceRequest = new ServiceRequest({
          reference,
          clientId: client._id,
          clientSnapshot: {
            fullName: client.fullName,
            email: client.email ?? null,
            phone: client.phone,
            organisation: client.organisation ?? null
          },
          items: requestItems,
          status: RequestStatus.NEW,
          submittedAt
        });

        await serviceRequest.save({ session });

        return {
          reference,
          submittedAt
        };
      }
    );

    if (!result) {
      throw new ApiError(
        500,
        "The service request could not be completed."
      );
    }

    return result;
  } finally {
    await session.endSession();
  }
}

import { z } from "zod";

const objectIdSchema = z
  .string()
  .regex(/^[0-9a-fA-F]{24}$/, "Invalid service ID.");

const serviceRequestItemSchema = z.object({
  serviceId: objectIdSchema,

  formVersion: z
    .number()
    .int()
    .min(1),

  answers: z.record(
    z.string(),
    z.unknown()
  )
}).strict();

export const createPublicServiceRequestSchema = z.object({
  body: z.object({
    client: z.object({
      fullName: z
        .string()
        .trim()
        .min(2, "Full name must contain at least 2 characters.")
        .max(100),

      email: z
        .email()
        .trim()
        .toLowerCase()
        .optional()
        .or(z.literal("")),

      phone: z
        .string()
        .trim()
        .min(7, "A valid phone number is required.")
        .max(20),

      organisation: z
        .string()
        .trim()
        .max(150)
        .optional()
    }).strict(),

    items: z
      .array(serviceRequestItemSchema)
      .min(1, "Select at least one service.")
  }).strict()
}).superRefine(({ body }, context) => {
  const serviceIds = body.items.map((item) => item.serviceId);

  if (new Set(serviceIds).size !== serviceIds.length) {
    context.addIssue({
      code: "custom",
      path: ["body", "items"],
      message: "A service can only be selected once per request."
    });
  }
});

export type CreatePublicServiceRequestInput = z.output<
  typeof createPublicServiceRequestSchema
>["body"];

import { Types } from "mongoose";

import { ApiError } from "../../shared/errors/ApiError.js";

import { Service } from "./Service.js";
import { ServiceCategory } from "./ServiceCategory.js";
import { ServiceForm } from "./ServiceForm.js";

async function getActiveServiceById(serviceId: string) {
  if (!Types.ObjectId.isValid(serviceId)) {
    throw new ApiError(400, "Invalid service ID.");
  }

  const service = await Service.findOne({
    _id: serviceId,
    isActive: true,
    isDeleted: false
  }).lean();

  if (!service) {
    throw new ApiError(404, "Service not found.");
  }

  const category = await ServiceCategory.findOne({
    _id: service.categoryId,
    isActive: true,
    isDeleted: false
  }).lean();

  if (!category) {
    throw new ApiError(404, "Service not found.");
  }

  return { service, category };
}

export async function getPublicServices() {
  const categories = await ServiceCategory.find({
    isActive: true,
    isDeleted: false
  })
    .sort({ displayOrder: 1, name: 1 })
    .lean();

  if (categories.length === 0) {
    return { categories: [] };
  }

  const services = await Service.find({
    categoryId: { $in: categories.map((category) => category._id) },
    isActive: true,
    isDeleted: false
  })
    .sort({ displayOrder: 1, name: 1 })
    .lean();

  return {
    categories: categories.map((category) => ({
      ...category,
      services: services.filter(
        (service) =>
          service.categoryId.toString() === category._id.toString()
      )
    }))
  };
}

export async function getPublicService(serviceId: string) {
  return getActiveServiceById(serviceId);
}

export async function getPublicServiceForm(serviceId: string) {
  const { service } = await getActiveServiceById(serviceId);

  const form = await ServiceForm.findOne({
    serviceId: service._id,
    isActive: true
  })
    .select("serviceId version fields publishedAt")
    .lean();

  if (!form) {
    throw new ApiError(404, "No published form is available for this service.");
  }

  return {
    serviceId: form.serviceId,
    version: form.version,
    publishedAt: form.publishedAt,
    fields: form.fields
      .slice()
      .sort((a, b) => a.displayOrder - b.displayOrder)
      .map((field) => {
        const {
          acceptedMimeTypes,
          maxFiles,
          maxFileSizeBytes,
          ...publicField
        } = field;

        return publicField;
      })
  };
}

function createQuotationNumber(): string {
  const year = new Date().getFullYear();

  const reference = randomBytes(5)
    .toString("hex")
    .toUpperCase();

  return `PA-QUO-${year}-${reference}`;
}

function hashToken(token: string): string {
  return createHash("sha256")
    .update(token)
    .digest("hex");
}

function validateObjectId(
  value: string,
  field: string
): Types.ObjectId {
  if (!Types.ObjectId.isValid(value)) {
    throw new ApiError(400, `Invalid ${field}.`);
  }

  return new Types.ObjectId(value);
}

function escapeRegex(value: string): string {
  return value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

function calculateQuotationAmounts(
  items: QuotationItemInput[],
  discountKobo: number
) {
  if (items.length === 0) {
    throw new ApiError(
      400,
      "A quotation must contain at least one item."
    );
  }

  if (
    !Number.isSafeInteger(discountKobo) ||
    discountKobo < 0
  ) {
    throw new ApiError(400, "Invalid quotation discount.");
  }

  const calculatedItems = items.map((item) => {
    if (
      !Number.isSafeInteger(item.quantity) ||
      item.quantity < 1 ||
      !Number.isSafeInteger(item.unitAmountKobo) ||
      item.unitAmountKobo < 0
    ) {
      throw new ApiError(
        400,
        "Invalid quotation item amount or quantity."
      );
    }

    const totalAmountKobo =
      item.quantity * item.unitAmountKobo;

    if (!Number.isSafeInteger(totalAmountKobo)) {
      throw new ApiError(
        400,
        "Quotation item total exceeds the supported amount."
      );
    }

    return {
      serviceRequestItemId: item.serviceRequestItemId
        ? validateObjectId(
            item.serviceRequestItemId,
            "service request item ID"
          )
        : null,

      serviceName: item.serviceName.trim(),
      description: item.description.trim(),
      deliverables: item.deliverables.map(
        (value) => value.trim()
      ),
      quantity: item.quantity,
      unitAmountKobo: item.unitAmountKobo,
      totalAmountKobo
    };
  });

  const subtotalKobo = calculatedItems.reduce(
    (total, item) => total + item.totalAmountKobo,
    0
  );

  if (!Number.isSafeInteger(subtotalKobo)) {
    throw new ApiError(
      400,
      "Quotation subtotal exceeds the supported amount."
    );
  }

  if (discountKobo > subtotalKobo) {
    throw new ApiError(
      400,
      "Quotation discount cannot exceed the subtotal."
    );
  }

  return {
    items: calculatedItems,
    subtotalKobo,
    discountKobo,
    totalAmountKobo: subtotalKobo - discountKobo
  };
}

async function validateQuotationItems(
  serviceRequestId: Types.ObjectId,
  items: QuotationItemInput[],
  session: mongoose.ClientSession
) {
  const serviceRequest = await ServiceRequest.findOne({
    _id: serviceRequestId,
    isDeleted: false
  }).session(session);

  if (!serviceRequest) {
    throw new ApiError(404, "Service request not found.");
  }

  const requestItemIds = new Set(
    serviceRequest.items.map(
      (item) =>
        (
          item as typeof item & {
            _id: Types.ObjectId;
          }
        )._id.toString()
    )
  );

  for (const item of items) {
    if (
      item.serviceRequestItemId &&
      !requestItemIds.has(item.serviceRequestItemId)
    ) {
      throw new ApiError(
        400,
        "A quotation item references a service that does not belong to this request."
      );
    }
  }

  return serviceRequest;
}

function getTokenExpiry(validUntil: Date): Date {
  const defaultExpiry = new Date(
    Date.now() + ACCESS_TOKEN_DURATION_MS
  );

  return validUntil < defaultExpiry
    ? validUntil

async function findValidAccessToken(
  token: string,
  session?: mongoose.ClientSession
) {
  if (!token || token.length < 32 || token.length > 512) {
    throw new ApiError(
      404,
      "Quotation link is invalid or unavailable."
    );
  }

  const tokenDocument = await QuotationAccessToken.findOne({
    tokenHash: hashToken(token),
    revokedAt: null,
    expiresAt: { $gt: new Date() }
  })
    .select("+tokenHash")
    .session(session ?? null);

  if (!tokenDocument) {
    throw new ApiError(
      404,
      "Quotation link is invalid or unavailable."
    );
  }

  return tokenDocument;
}

/**
 * Update an existing quotation draft.
 */
export async function updateQuotationDraft(
  quotationId: string,
  input: UpdateQuotationInput,
  adminId: string,
  context: AuditContext = {}
) {
  const id = validateObjectId(
    quotationId,
    "quotation ID"
  );

  const administratorId = validateObjectId(
    adminId,
    "administrator ID"
  );

  const session = await mongoose.startSession();

  try {
    await session.withTransaction(async () => {
      const quotation = await Quotation.findById(id)
        .session(session);

      if (!quotation) {
        throw new ApiError(404, "Quotation not found.");
      }

      if (quotation.status !== QuotationStatus.DRAFT) {
        throw new ApiError(
          409,
          "Only draft quotations can be edited."
        );
      }

      const changes: Record<string, unknown> = {};

      if (input.items !== undefined) {
        await validateQuotationItems(
          quotation.serviceRequestId,
          input.items,
          session
        );

        const amounts = calculateQuotationAmounts(
          input.items,
          input.discountKobo ?? quotation.discountKobo
        );

        quotation.items = amounts.items as typeof quotation.items;
        quotation.subtotalKobo = amounts.subtotalKobo;
        quotation.discountKobo = amounts.discountKobo;
        quotation.totalAmountKobo = amounts.totalAmountKobo;

        changes.itemsUpdated = true;
        changes.totalAmountKobo = amounts.totalAmountKobo;
      } else if (input.discountKobo !== undefined) {
        const amounts = calculateQuotationAmounts(
          quotation.items.map((item) => ({
            serviceRequestItemId:
              item.serviceRequestItemId?.toString() ?? null,
            serviceName: item.serviceName,
            description: item.description,
            deliverables: item.deliverables,
            quantity: item.quantity,
            unitAmountKobo: item.unitAmountKobo
          })),
          input.discountKobo
        );

        quotation.discountKobo = amounts.discountKobo;
        quotation.totalAmountKobo = amounts.totalAmountKobo;

        changes.discountKobo = amounts.discountKobo;
        changes.totalAmountKobo = amounts.totalAmountKobo;
      }

      if (input.scope !== undefined) {
        quotation.scope = input.scope.trim();
        changes.scopeUpdated = true;
      }

      if (input.terms !== undefined) {
        quotation.terms = input.terms.map(
          (term) => term.trim()
        );

        changes.termsUpdated = true;
      }

      if (input.validUntil !== undefined) {
        const validUntil = new Date(input.validUntil);

        if (
          Number.isNaN(validUntil.getTime()) ||
          validUntil <= new Date()
        ) {
          throw new ApiError(
            400,
            "Quotation validity must be a future date."
          );
        }

        quotation.validUntil = validUntil;
        changes.validUntil = validUntil.toISOString();
      }

      await quotation.save({ session });

      await writeAuditLog(
        {
          adminId: administratorId,
          action: "quotation.updated",
          resourceId: quotation._id,
          description: "Quotation draft updated.",
          changes
        },
        context,
        session
      );
    });

    const updatedQuotation = await Quotation.findById(id);

    if (!updatedQuotation) {
      throw new ApiError(404, "Quotation not found.");
    }

    return updatedQuotation;
  } finally {
    await session.endSession();
  }
}

/**
 * Issue a quotation and generate its secure client access token.
 */
export async function issueQuotation(
  quotationId: string,
  adminId: string,
  context: AuditContext = {}
) {
  const id = validateObjectId(
    quotationId,
    "quotation ID"
  );

  const administratorId = validateObjectId(
    adminId,
    "administrator ID"
  );

  const session = await mongoose.startSession();

  let issuedQuotationId: Types.ObjectId | null = null;
  let rawToken: string | null = null;

  try {
    await session.withTransaction(async () => {
      const quotation = await Quotation.findById(id)
        .session(session);

      if (!quotation) {
        throw new ApiError(404, "Quotation not found.");
      }

      if (quotation.status !== QuotationStatus.DRAFT) {
        throw new ApiError(
          409,
          "Only draft quotations can be issued."
        );
      }

      if (quotation.validUntil <= new Date()) {
        throw new ApiError(
          400,
          "The quotation validity date has passed."
        );
      }

      if (
        quotation.items.length === 0 ||
        quotation.totalAmountKobo < 0
      ) {
        throw new ApiError(
          400,
          "Quotation must contain valid items and amounts."
        );
      }

      const client = await Client.findOne({
        _id: quotation.clientId,
        isDeleted: false
      }).session(session);

      const serviceRequest = await ServiceRequest.findOne({
        _id: quotation.serviceRequestId,
        isDeleted: false
      }).session(session);

      if (!client || !serviceRequest) {
        throw new ApiError(
          409,
          "The associated client or service request is unavailable."
        );
      }

      const token = randomBytes(32).toString("hex");

      const accessToken = new QuotationAccessToken({
        quotationId: quotation._id,
        quotationNumber: quotation.quotationNumber,
        quotationVersion: quotation.version,
        clientId: quotation.clientId,
        tokenHash: hashToken(token),
        expiresAt: getTokenExpiry(quotation.validUntil)
      });

      await accessToken.save({ session });

      quotation.status = QuotationStatus.SENT;
      quotation.issuedAt = new Date();

      await quotation.save({ session });

      await writeAuditLog(
        {
          adminId: administratorId,
          action: "quotation.issued",
          resourceId: quotation._id,
          description: "Quotation issued to client.",
          metadata: {
            quotationNumber: quotation.quotationNumber,
            quotationVersion: quotation.version,
            accessTokenId: accessToken._id.toString(),
            tokenExpiresAt: accessToken.expiresAt.toISOString()
          }
        },
        context,
        session
      );

      issuedQuotationId = quotation._id;
      rawToken = token;
    });

    if (!issuedQuotationId || !rawToken) {
      throw new ApiError(
        500,
        "Quotation issuance could not be completed."
      );
    }

    const quotation = await Quotation.findById(
      issuedQuotationId
    );

    if (!quotation) {
      throw new ApiError(
        500,
        "Issued quotation could not be retrieved."
      );
    }

    return {
      quotation,
      accessToken: rawToken
    };
  } finally {
    await session.endSession();
  }
}

/**
 * List quotations for the administrative interface.
 *
 * The aggregation excludes quotations whose associated
 * clients or service requests have been soft-deleted.
 */
export async function listQuotations(
  query: ListQuotationsQuery
) {
  const {
    page,
    limit,
    search,
    status,
    clientId,
    serviceRequestId,
    from,
    to,
    sortBy,
    sortOrder
  } = query;

  const match: Record<string, unknown> = {};

  if (status) {
    match.status = status;
  }

  if (clientId) {
    match.clientId = validateObjectId(
      clientId,
      "client ID"
    );
  }

  if (serviceRequestId) {
    match.serviceRequestId = validateObjectId(
      serviceRequestId,
      "service request ID"
    );
  }

  if (from || to) {
    const createdAt: {
      $gte?: Date;
      $lte?: Date;
    } = {};

    if (from) {
      createdAt.$gte = new Date(from);
    }

    if (to) {
      createdAt.$lte = new Date(to);
    }

    match.createdAt = createdAt;
  }

  if (search) {
    const searchExpression = new RegExp(
      escapeRegex(search),
      "i"
    );

    match.$or = [
      { quotationNumber: searchExpression },
      { serviceRequestReference: searchExpression },
      { "clientSnapshot.fullName": searchExpression },
      { "clientSnapshot.email": searchExpression },
      { "clientSnapshot.phone": searchExpression },
      { "items.serviceName": searchExpression }
    ];
  }

  const sortDirection: 1 | -1 =
    sortOrder === "asc" ? 1 : -1;

  const sort: Record<string, 1 | -1> = {
    [sortBy]: sortDirection,
    _id: -1
  };

  const skip = (page - 1) * limit;

  const pipeline: mongoose.PipelineStage[] = [
    {
      $match: match
    },
    {
      $lookup: {
        from: Client.collection.name,
        localField: "clientId",
        foreignField: "_id",
        pipeline: [
          {
            $match: {
              isDeleted: false
            }
          },
          {
            $project: {
              _id: 1
            }
          }
        ],
        as: "activeClient"
      }
    },
    {
      $lookup: {
        from: ServiceRequest.collection.name,
        localField: "serviceRequestId",
        foreignField: "_id",
        pipeline: [
          {
            $match: {
              isDeleted: false
            }
          },
          {
            $project: {
              _id: 1
            }
          }
        ],
        as: "activeServiceRequest"
      }
    },
    {
      $match: {
        activeClient: { $ne: [] },
        activeServiceRequest: { $ne: [] }
      }
    },
    {
      $facet: {
        quotations: [
          {
            $sort: sort
          },
          {
            $skip: skip
          },
          {
            $limit: limit
          },
          {
            $project: {
              activeClient: 0,
              activeServiceRequest: 0
            }
          }
        ],
        total: [
          {
            $count: "count"
          }
        ]
      }
    }
  ];

  const [result] = await Quotation.aggregate<QuotationListAggregationResult>(
    pipeline
  );

  const quotations = result?.quotations ?? [];
  const total = result?.total[0]?.count ?? 0;

  return {
    quotations,
    page,
    limit,
    total,
    totalPages: Math.ceil(total / limit)
  };
}

/**
 * Record a client's response to an issued quotation.
 */
export async function respondToQuotation(
  token: string,
  input: ClientResponseInput,
  context: AuditContext = {}
) {
  const session = await mongoose.startSession();

  let quotationId: Types.ObjectId | null = null;

  try {
    await session.withTransaction(async () => {
      const accessToken = await findValidAccessToken(
        token,
        session
      );

      const quotation = await Quotation.findOne({
        _id: accessToken.quotationId,
        quotationNumber: accessToken.quotationNumber,
        version: accessToken.quotationVersion
      }).session(session);

      if (
        !quotation ||
        quotation.status !== QuotationStatus.SENT
      ) {
        throw new ApiError(
          404,
          "Quotation link is invalid or unavailable."
        );
      }

      if (quotation.validUntil <= new Date()) {
        throw new ApiError(
          410,
          "This quotation has expired."
        );
      }

      const responseStatus = {
        approved: QuotationStatus.APPROVED,
        declined: QuotationStatus.DECLINED,
        revision_requested: QuotationStatus.REVISION_REQUESTED
      } as const;

      const respondedAt = new Date();

      const [quotationResponse] = await QuotationResponse.create(
        [
          {
            quotationId: quotation._id,
            quotationNumber: quotation.quotationNumber,
            quotationVersion: quotation.version,
            response: input.response,
            comments: input.comments ?? null,
            accessTokenId: accessToken._id,
            respondedAt
          }
        ],
        { session }
      );

      quotation.status = responseStatus[input.response];
      quotation.respondedAt = respondedAt;

      await quotation.save({ session });

      await QuotationAccessToken.updateMany(
        {
          quotationId: quotation._id,
          quotationVersion: quotation.version,
          revokedAt: null
        },
        {
          $set: { revokedAt: respondedAt }
        },
        { session }
      );

      await writeAuditLog(
        {
          adminId: null,
          action: `quotation.client_${input.response}`,
          resourceId: quotation._id,
          description: `Client ${input.response.replaceAll("_", " ")} the quotation.`,
          metadata: {
            quotationNumber: quotation.quotationNumber,
            quotationVersion: quotation.version,
            responseId: quotationResponse?._id.toString() ?? null
          }
        },
        context,
        session
      );

      quotationId = quotation._id;
    });

    if (!quotationId) {
      throw new ApiError(
        500,
        "Quotation response could not be recorded."
      );
    }

    return Quotation.findById(quotationId);
  } finally {
    await session.endSession();
  }
}
